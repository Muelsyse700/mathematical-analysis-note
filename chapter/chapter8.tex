\chapter{空间解析几何}
\section{向量与坐标系}
\subsection{向量的定义与性质}
\begin{definition}
    空间中既有大小又有方向的量称为\overtext{向量}{vector}，记作$\vb{v}$.它的大小记为$|\vb{v}|$.
    \begin{itemize}
        \item 对于两个大小相等、方向相同的向量$\vb{a},\vb{b}$，称二者相等，记为$\vb{a} = \vb{b}$.
        \item 大小为$0$的向量称为\overtext{零向量}{zero vector}，记作$\vb{0}$，它没有确定的方向.
        \item 大小相等而方向相反的向量称为\overtext{负向量}{negative vector}，记作$-\vb{v}$.
        \item 大小为$1$的向量称为\overtext{单位向量}{unit vector}，记为$\vb{e_v} = \dfrac{\vb{v}}{|\vb{v}|}$.
    \end{itemize}
\end{definition}

\begin{enumerate}[property]
    \item 向量具有加法且具备如下性质：
    \begin{itemize}
        \item \textbf{交换律：}$\vb{a + b = b + a}$.
        \item \textbf{结合律：}$\vb{(a + b) + c = a + (b + c)}$.
        \item \textbf{零元与逆元：}$\vb{a + 0 = a},\quad \vb{a + (-a) = 0}$.
    \end{itemize}
    上述性质利用平行四边形法则或三角形法则易证.
    \item 向量具有数乘.设$\lambda,\mu \in \mathbb{R}$，满足：
    \begin{itemize}
        \item $\lambda(\vb{a+b}) = \lambda\vb{a} + \lambda\vb{b}$.
        \item $(\lambda + \mu)\vb{a} = \lambda\vb{a} + \mu\vb{a}$.
        \item $(\lambda\mu)\vb{a} = \lambda(\mu\vb{a})$.
    \end{itemize}
\end{enumerate}

\begin{theorem}
    设$\vb{a,b}$是两个向量，则：
    \[\vb{|a+b| \leqslant |a| + |b|}\]
    当且仅当存在非负实数$\lambda$满足$\vb{a} = \lambda\vb{b}$，等号成立.
\end{theorem}


\subsection{向量的共线与共面}
\begin{definition}
    如果通过平移：
    \begin{itemize}
        \item 能使一组向量$\vb{a,b}$同处一条直线上，那么称它们是\textbf{共线}或\overtext{平行}{parallel}的，记为$\vb{a}\parallel\vb{b}$.
        \item 能使一组向量$\vb{a,b,c}$同处一个平面上，那么称它们是\textbf{共面}的.
    \end{itemize}
\end{definition}

\begin{theorem}
    \begin{itemize}
        \item 两个向量$\vb{a,b}$共线，当且仅当它们线性相关，即存在不全为零的实数$\lambda,\mu$满足：
        \[\lambda\vb{a} + \mu\vb{b} = \vb{0}\]
        \item 三个向量$\vb{a,b,c}$共面，当且仅当它们线性相关，即存在三个不全为零的实数$\lambda,\mu,\nu$满足：
        \[\lambda\vb{a} + \mu\vb{b} + \nu\vb{c} = \vb{0}\]
    \end{itemize}
\end{theorem}

\begin{proof}
    以两个向量$\vb{a,b}$的证明为例：

    \textbf{充分性：}设非零向量$\vb{a,b}$共线，即：
    \[\vb{a} = \pm k \vb{b},\quad k = \frac{|\vb{a}|}{|\vb{b}|} \neq 0 \quad\Rightarrow\quad \vb{a} \mp k \vb{b} = \vb{0}\]
    取$\lambda = 1 \neq 0,\mu = \mp k \neq 0$，满足：
    \[\lambda \vb{a} + \mu \vb{b} = \vb{0}\]

    \textbf{必要性：}若存在不全为零的数$\lambda,\mu$使得：
    \[\lambda \vb{a} + \mu \vb{b} = \vb{0}\]
    不妨设$\lambda \neq 0$，则有：
    \[\vb{a} = \frac{\mu}{\lambda}\vb{b}\]
    满足共线的定义.

    对于多个向量的证明，只需要对向量进行组合，构造共线向量即可.
\end{proof}


\subsection{向量的内积和外积}
\subsubsection{内积}
设$\vb{a,b}$是两个向量，它们的夹角定义为$\theta = \theta(\vb{a,b}) \in [0,\pi]$，显然有$\theta(\vb{a,b}) = \theta(\vb{b,a})$.对于任意非零的实数$\lambda$满足：
\[\theta(\lambda\vb{a,b}) = \begin{cases}
    \theta(\vb{a,b}),\quad\lambda>0\\
    \pi-\theta(\vb{a,b}),\quad\lambda<0
\end{cases}\]

\begin{definition}
    定义两个向量$\vb{a,b}$的\overtext{内积}{inner product}\!\!\textbf{(\!\!\overtext{点乘}{dot product}\!\!、\overtext{数量积}{scalar product}\!\!)}为如下运算：
    \[\vb{a}\cdot\vb{b} = (\vb{a},\vb{b}) \coloneq |\vb{a}| |\vb{b}| \cos\theta(\vb{a,b}) \in \mathbb{R},\quad \theta \in [0,\pi]\]
    若其中一个向量是零向量，则规定点乘结果为$0$.
\end{definition}

显然，从定义上看，总是有：
\[\begin{cases}
    \vb{a \cdot b} > 0,\quad \theta(\vb{a,b}) < \dfrac{\pi}{2} \\[0.5em]
    \vb{a \cdot b} = 0,\quad \theta(\vb{a,b}) = \dfrac{\pi}{2} \quad\Rightarrow\quad \text{此时称$\vb{a,b}$\overtext{正交}{orthogonal}或\overtext{垂直}{vertical}，记作$\vb{a}\perp\vb{b}$}
\end{cases}\]
几何上，$|\vb{a}|\cos\theta$就是$\vb{a}$在$\vb{b}$所在直线上投影向量$\vb{a_b}$的有向长度，有如下关系式：
\[\vb{a_b} = (|\vb{a}|\cos\theta)\vb{e_b} = (\vb{a \cdot e_b})\vb{e_b}\]
上式表明当$\theta \in [\frac{\pi}{2},\pi]$时，投影向量$\vb{e_b}$与$\vb{b}$共线但反向.

于是点乘的含义可以理解为$\vb{a}$在$\vb{b}$方向上的投影向量的有向长度乘$\vb{b}$的长度，或是说与$\vb{b}$垂直的分量对$\vb{a \cdot b}$没有贡献.

点乘具有如下性质：
\begin{itemize}
    \item \textbf{交换律：}$\vb{a \cdot b = b \cdot a}$.
    \item \textbf{结合律：}$(\lambda \vb{a})\cdot \vb{b} = \lambda (\vb{a \cdot b})$.
    \item \textbf{分配律：}$\vb{(a + b)\cdot c = a \cdot c + b \cdot c}$
\end{itemize}


\subsubsection{外积}
对于不共面向量组$\{\vb{a,b,c}\}$，若其满足右手系，则$\{\vb{b,c,a}\},\{\vb{c,a,b}\}$仍是右手系，但$\{\vb{b,a,c}\}$是左手系.

\begin{definition}
    定义两个向量$\vb{a,b}$的\overtext{外积}{outer product}\!\!\textbf{(\!\!\overtext{叉乘}{cross product}\!\!、\overtext{向量积}{vector product}\!\!)}的结果为一个矢量，其大小规定为：
    \[|\vb{a \times b}| \coloneq |\vb{a}||\vb{b}|\sin\theta(\vb{a,b}),\quad \theta\in [0,\pi]\]
    方向规定为$\vb{a \times b}$与$\vb{a,b}$垂直，且$\{\vb{a,b,a \times b}\}$构成右手系.

    若其中一个向量是零向量，则规定叉乘结果也是零向量.
\end{definition}

显然，从定义上看，若是两个向量贡献，其叉乘结果是零向量.

从几何上看，外积$\vb{a \times b}$的大小即为以$\vb{a,b}$为邻边的平行四边形的面积.

叉乘具有如下性质：
\begin{itemize}
    \item $(\lambda\vb{a})\times\vb{b} = \vb{a}\times(\lambda\vb{b}) = \lambda(\vb{a}\times\vb{b})$.
    \item \textbf{反称性：}$\vb{a \times b = -a \times b}$.
    \item \textbf{分配律：}$\vb{a \times (b + c) = a \times b + a \times c}$.
\end{itemize}


\subsubsection{混合积}
\begin{definition}
    称$\vb{(a \times b)\cdot c}$为三个向量$\vb{a,b,c}$的\overtext{混合积}{mixed product}.
\end{definition}

从几何上看，混合积结果的绝对值是以$\vb{a,b,c}$为棱的平行六面体的体积.

若是有$\vb{(a \times b)\cdot c} = 0$，则表明$\vb{a,b,c}$共面.

混合积具有\overtext{轮换对称性}{cyclic symmetry}，即：
\[\vb{a \times b}\cdot c = \vb{b \times c}\cdot a = \vb{c \times a}\cdot b\]


\subsubsection{有向长度、面积和体积}
\begin{definition}
    有向长度、面积和体积即为具备正负性的长度、面积和体积，规定：
    \begin{itemize}
        \item 当$\theta(\vb{a,b})$为锐角时，有向长度$|\vb{a}|\cos\theta$为正；当$\theta(\vb{a,b})$为钝角时为负.
        \item 当$\{\vb{a,b,a \times b}\}$构成右手系时，有向面积$\vb{a \times b}$为正，反之构成左手系则为负.
        \item 当$\{\vb{a,b,c}\}$构成右手系时，有向体积$\vb{(a \times b)\cdot c}$为正，反之构成左手系则为负.
    \end{itemize}
\end{definition}


\subsection{向量的坐标表示与空间坐标系}
\begin{definition}
    设集合$V$由全部空间向量构成的集合，在其上定义了向量的加法和数乘，称$V$是一个\overtext{线性空间}{linear space}\textbf{(向量空间)}.

    称$V$中三个不共面的向量组$\{\vb{e}_1,\vb{e}_2,\vb{e_3}\}$为$V$的一组\overtext{基向量}{basis vectors}，简称\overtext{基}{base}.

    选定一个原点$O$，称$\mathbb{F} = [O;\vb{e}_1,\vb{e}_2,\vb{e}_3]$为$V$中的一个\overtext{仿射坐标系}{affine coordinate system}，其中一向量$\vb{x}$对应的坐标$(x_1,x_2,x_3)$称为\overtext{仿射坐标}{affine coordinates}.
\end{definition}

\begin{theorem}
    若线性空间$V$中存在一组基$\{\vb{e}_1,\vb{e}_2,\vb{e}_3\}$，则对$\forall \vb{x} \in V$，都存在唯一的一组数$(x_1,x_2,x_3)$使得：
    \[\vb{x} = x_1\vb{e}_1 + x_2\vb{e}_2 + x_3\vb{e}_3 \quad\text{或}\quad \vb{x} \mapsto (x_1,x_2,x_3)\]
\end{theorem}

\begin{definition}
    取线性空间$V$中的$\{\vb{i,j,k}\}$是两两正交且形成右手系的单位向量，称他们为$V$的\overtext{标准正交基}{orthonormal basis}，称$\mathbb{F} = [O;\vb{i,j,k}]$为\overtext{直角坐标系}{cartesian coordinate system}.
\end{definition}

直角坐标系中满足如下等式：
\begin{gather*}
    \vb{i \cdot i} = \vb{j \cdot j} = \vb{k \cdot k} = 1 \\
    \vb{i \cdot j} = \vb{j \cdot k} = \vb{k \cdot i} = 0 \\
    \vb{i \times i} =\vb{j \times j} = \vb{k \times k} = \vb{0} \\
    \vb{i \times j} = \vb{k},\quad \vb{j \times k} = \vb{i},\quad \vb{k \times i} = \vb{j}
\end{gather*}
根据上述内容不难推出内积、外积和混合积的坐标表示：
\[\vb{a \cdot b} = a_1b_1 + a_2b_2 + a_3b_3,\quad \vb{a \times b} = \begin{vmatrix}
    \vb{i} & \vb{j} & \vb{k} \\
    a_1 & a_2 & a_3 \\
    b_1 & b_2 & b_3
\end{vmatrix},\quad \vb{(a \times b)\cdot c} = \begin{vmatrix}
    a_1 & a_2 & a_3 \\
    b_1 & b_2 & b_3 \\
    c_1 & c_2 & c_3
\end{vmatrix}\]

\begin{theorem}
    \textbf{Cauchy-Schwarz不等式}
    \[(a_1b_1 + a_2b_2 + a_3b_3)^2 \leqslant (a_1^2 + a_2^2 + a_3^2)(b_1^2 + b_2^2 + b_3^2)\]
\end{theorem}

\begin{proof}
    设有$\vb{a} = (a_1,a_2,a_3),\vb{b}=(b_1,b_2,b_3)$，则不等式等价为：
    \[(\vb{a \cdot b})^2 = (|\vb{a}||\vb{b}|\cos\theta)^2 \leqslant |\vb{a}|^2 |\vb{b}|^2\]
    显然得证.
\end{proof}

\begin{definition}
    在空间中取定一点$O$，则空间中任意一点$P$对应一个向量$\va{OP}$，称为$P$的\overtext{位置向量}{positional vector}，记作$\vb{r}(P)$，在直角坐标系下具有如下对应关系：
    \[P \leftrightarrow \va{OP} = x\vb{i} + y\vb{j} + z\vb{k} \leftrightarrow (x,y,z)\]
    $\va{OP}$的方向可由$\va{OP}$与三个坐标轴的夹角，即\overtext{方向角}{direction angles}来刻画：
    \[x = |\va{OP}|\cos\alpha,\quad y = |\va{OP}|\cos\beta,\quad z = |\va{OP}|\cos\gamma\]
    称方向角的余弦为\overtext{方向余弦}{direction cosines}.
\end{definition}
注意到，$\va{OP}$的方向余弦构成了$\va{OP}$方向上的一个单位向量$(\cos\alpha,\cos\beta,\cos\gamma)$.


\section{坐标变换和其它常用坐标系}
\subsection{坐标变换}
\subsubsection{平移}
将坐标系$\mathbb{F}$平移得到的新坐标系$\mathbb{F'}$，平移向量为$\va{OO'} = a\vb{i} + b\vb{j} + c\vb{k}$.设$P$在两个坐标系下的位置向量为：
\[\va{OP} = x\vb{i} + y\vb{j} + z\vb{k},\qquad \va{O'P} = x'\va{i} + y'\va{j} + z'\va{k}\]
则根据$\va{O'P} = \va{OP} - \va{OO'}$可得平移后的坐标：
\[x'=x-a,\quad y'=y-b,\quad z'=z-c\]


\subsubsection{旋转}
假设$\mathbb{F'}$由$\mathbb{F}$保持原点不动旋转得到，设两个坐标系基向量两两之间的夹角为：
\begin{gather*}
    \begin{NiceArray}{c|ccc}
        & \vb{i} & \vb{j} & \vb{k} \\
        \Hline
        \vb{i'} & \alpha_1 & \beta_1 & \gamma_1 \\
        \vb{j'} & \alpha_2 & \beta_2 & \gamma_2 \\
        \vb{k'} & \alpha_3 & \beta_3 & \gamma_3 \\
    \end{NiceArray} \quad\Rightarrow\quad \begin{cases}
        \vb{i'} = \cos\alpha_1\vb{i} + \cos\beta_1\vb{j} + \cos\gamma_1\vb{k} \\
        \vb{j'} = \cos\alpha_2\vb{i} + \cos\beta_2\vb{j} + \cos\gamma_2\vb{k} \\
        \vb{k'} = \cos\alpha_3\vb{i} + \cos\beta_3\vb{j} + \cos\gamma_3\vb{k}
    \end{cases}\\
    \begin{bmatrix}
        \cos\alpha_1 & \cos\beta_1 & \cos\gamma_1 \\
        \cos\alpha_2 & \cos\beta_2 & \cos\gamma_2 \\
        \cos\alpha_3 & \cos\beta_3 & \cos\gamma_3
    \end{bmatrix} \begin{bmatrix}
        \vb{i} \\ \vb{j} \\ \vb{k}
    \end{bmatrix} =  \begin{bmatrix}
        \vb{i'} \\ \vb{j'} \\ \vb{k'}
    \end{bmatrix} 
\end{gather*}
也就是说，左边的$3 \times 3$矩阵实现了旋转变换操作.


\subsection{其它坐标系}
\begin{itemize}
    \item \textbf{极坐标：}极坐标$(r,\theta)$与直角坐标$(x,y)$之间的变换关系为：
    \[\begin{cases}
        x = r\cos\theta \\
        y = r\sin\theta
    \end{cases} \Leftrightarrow\quad \begin{cases}
        r = \sqrt{x^2 + y^2} \\
        \theta = \arctan\dfrac{y}{x}
    \end{cases}\]
    \item \textbf{柱坐标：}柱坐标$(r,\theta,z)$与直角坐标$(x,y,z)$之间的变换关系为：
    \[x = r\cos\theta,\quad y = r\sin\theta,\quad z = z\]
    \item \textbf{球坐标：}球坐标$(r,\theta,\varphi)$与直角坐标$(x,y,z)$之间的变换关系为：
    \[x = r\sin\theta\cos\varphi,\quad y = r\sin\theta\sin\varphi,\quad z = r\cos\theta\]
\end{itemize}


\section{平面与直线}
\subsection{平面方程}
\subsubsection{平面的一般方程}
平面的一般方程来自于如下方式的构造：在给定一点$P_0(x_0,y_0,z_0)$上延伸出一个向量$\vb{n} = (a,b,c)$，取一个平面过点$P_0$且垂直于这个向量.

按照上述构造的过程，取平面上任意一点$P(x,y,z)$，必然有$\vb{n}\cdot\va{P_0P} = 0$，即：
\[a(x-x_0) + b(y-y_0) + c(z-z_0) = 0\]
令$d \equiv -(ax_0 + by_0 + cz_0)$，即：
\[ax + by + cz + d = 0\]
满足上式的点$(x,y,z)$一定在上述平面上，上式称为\textbf{平面的一般方程}.

上式也意味着，对于表示为平面的一般方程的平面，取$x,y,z$的系数即位其法向量.


\subsubsection{平面的截距式方程}
平面的截距式方程来自于如下方式的构造：对于不共线的三个点$P_1,P_2,P_3$，必有$\va{P_1P_2},\va{P_1P_3}$形成一个平面，所以由外积可以得到这个平面点法向量$\va{P_1P_2}\times\va{P_1P_3}$.

取平面上任意一点$P(x,y,z)$，总是有：
\[(\va{P_1P_2} \times \va{P_1P_3}) \cdot \va{P_1P} = \begin{vmatrix}
    x_2 - x_1 & y_2-y_1 & z_2-z_1 \\
    x_3 - x_1 & y_3-y_1 & z_3-z_1 \\
    x - x_1 & y-y_1 & z-z_1
\end{vmatrix} = 0\]
特别地，若是取三个坐标轴上的点$(\alpha,0,0),(0,\beta,0),(0,0,\gamma)$构造平面，上述方程简化为\textbf{平面的截距式方程}：
\[\frac{x}{\alpha} + \frac{y}{\beta} + \frac{z}{\gamma} = 1\]


\subsubsection{两平面的关系}
平面间相交与平行的关系主要是通过法向量之间的关系来确定的：
\begin{itemize}
    \item \textbf{平行：}若两平面平行，则有$\vb{n}_1 \parallel \vb{n}_2$.即存在$\lambda \neq 0$使得$\vb{n}_1 = \lambda \vb{n}_2$.
    
    对于平面的一般方程，有$a_1 = \lambda a_2, b_1 = \lambda b_2, c_1 = \lambda c_2$，若同时又有$d_1 = \lambda d_2$，则两平面重合.
    \item \textbf{相交：}两平面的法向量不平行，则两平面相交，相交的二面角为：
    \[\theta = \arccos\frac{|\vb{n}_1 \cdot \vb{n}_2|}{|\vb{n}_1||\vb{n}_2|}\]
\end{itemize}


\subsubsection{点到平面的距离}
设有平面外一点$P_0(x_0,y_0,z_0)$，过点$P_0$作平面的垂线，垂足为$P_1$.在平面上任取一点$P(x,y,z)$，设$\va{PP_0}$与平面法向量$\vb{n} = (a,b,c)$的夹角为$\theta$，则有：
\begin{align*}
    |\va{P_0P_1} &= |\va{PP_0}||\cos\theta| = \left|\frac{\vb{n}}{|\vb{n}|}\cdot\va{PP_0}\right| = \frac{|a(x_0-x) + b(y_0-y) + c(z_0-z)|}{\sqrt{a^2 + b^2 + c^2}}\\
    &= \frac{|ax_0 + by_0 + cz_0 +d|}{\sqrt{a^2 + b^2 + c^2}}
\end{align*}
其分子相当于把平面外一点的坐标代入平面方程.

空间中的一张平面将空间分为平面法向所指的一侧(上方)、平面本身、法向所指的另一侧(下方).判断一个点属于哪个部分，主要是看$\va{PP_0}$与法向量$\vb{n}$的夹角：
\[\vb{n}\cdot\va{PP_0} = ax_0 + by_0 + cz_0 + d\begin{cases}
    >0 \quad\Rightarrow\quad \text{是锐角，$P_0$在平面的上方.}\\
    =0 \quad\Rightarrow\quad \text{是直角，$P_0$在平面内.}\\
    <0 \quad\Rightarrow\quad \text{是钝角，$P_0$在平面的下方.}
\end{cases}\]
