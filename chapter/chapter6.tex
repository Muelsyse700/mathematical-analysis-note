\chapter{常微分方程初步}
\begin{definition}
    \overtext{微分方程}{differential equation}指的是联系着一个或多个自变量与函数及其各阶微商的关系式：
    \[F(x,y,y',y'',\cdots,y^{(n)}) = 0\]
    特别对只有单变量的形式称为\overtext{常微分方程}{ordinary differential equation}\textbf{(ODE)}；若方程关于$y,y,\cdots,y^{(n)}$均是一次的，则称之为$n$阶\overtext{线性微分方程}{linear differential equation}.
\end{definition}

\begin{definition}
    设有微分方程$F(x,y,y',y'',\cdots,y^{(n)}) = 0$，若有一函数族：
    \[y = \varphi(x;C_1,C_2,\cdots,C_n)\]
    包含该方程在某一区间上的所有解，其中$C_1,C_2,\cdots,C_n$为任意常数，则称该函数族为该微分方程的\overtext{通解}{general solution}.

    通过在通解中取定任意常数：
    \[C_1 = c_1,\quad C_2=c_2,\quad \cdots, \quad C_n=c_n\]
    所得到的一个具体解，称为该微分方程的\overtext{特解}{particular solution}.
\end{definition}

\section{一阶微分方程}
\subsection{分离变量法}
对于形如$\displaystyle\dv{y}{x}=f(x,y)$的微分方程，若是有：
\[\dv{y}{x} = f(x,y) = g(x)h(y)\]
其中$g,h$都是连续函数且$h(y)$不恒为$0$，则原方程可化为：
\[\frac{1}{h(y)}\dd{y} = g(x)\dd{x}\]
两边积分：
\[\int\frac{1}{h(y)}\dd{y} = \int g(x)\dd{x} \quad\Rightarrow\quad H(y) = G(x) + C\]
上述结果即为原微分方程的通解.特别地，若有$y_0$使得$h(y_0) = 0$，这也是原微分方程的一个解，但是这个解在进行分离变量的时候丢失，从而无法被通解所描述，这样的解称为\textbf{例外解}.

\begin{question}
    求解方程：$(1+x^2)y\dd{y} + \sqrt{1-y^2}\dd{x} = 0$.
\end{question}

\begin{solution}
    当$1-y^2 \neq 0$时，分离变量并积分：
    \begin{align*}
        \int\frac{y}{\sqrt{1-y^2}}\dd{y} &= -\int\frac{1}{1+x^2}\dd{x}\\
        \sqrt{1-y^2} &= \arctan x + C
    \end{align*}
    同时注意到，若有$1-y^2=0$，即$y=\pm1$也满足原微分方程，故而这是例外解.于是这个方程的全部解为：
    \[\sqrt{1-y^2} = \arctan x + C \quad\text{及}\quad y = \pm 1\]
\end{solution}


\subsection{齐次方程}
对于齐次方程，不能直接分离变量，但是经过特定步骤的换元后，就可以通过分离变量法求解.

\begin{definition}
    称$f(x,y)$是$n$次\overtext{齐次函数}{homogeneous function}，若其满足：
    \[f(tx,ty) = t^n f(x,y)\]
    称下述由同次齐次函数$P(x,y),Q(x,y)$构成的满足$f(tx,ty)=f(x,y)$的一阶微分方程为\textbf{齐次方程}：
    \[P(x,y)\dd{x} + Q(x,y)\dd{y} = 0\]
\end{definition}

所谓齐次，就是每一项中所有变量的次数之和一致，它们具有一致的缩放规律，即对输入放大$t$倍，则输出放大$t^n$倍.

从几何上看，如果把输入的$(x,y)$视作一个向量：
\[(x,y) = x \left(1,\frac{y}{x}\right)\]
其中$x$描述的是尺度，$\left(1,\dfrac{y}{x}\right)$则描述了方向，若$f$是$n$次齐次函数，则：
\[f(x,y) = x^n f\left(1,\frac{y}{x}\right) \xlongequal{\varphi(u)\equiv f(1,u)} x^n \varphi(u)\]
上式中$\varphi\left(\frac{y}{x}\right)$实际上就是一个控制单位矢量方向的函数.

如是而言，一个齐次函数可以被看作是：
\[\text{尺度}^n \times \text{仅与方向有关的函数}\]
那么对于$0$次齐次函数，它是一个仅对方向敏感的函数；对于$1$次齐次函数，其是一个对一维尺度和方向敏感的函数，以此类推.

对于齐次方程，由于有$\dfrac{P(x,y)}{Q(x,y)}$是$0$次齐次函数，因此可以将齐次方程化为：
\[\dv{y}{x} = \varphi\left(\frac{y}{x}\right) \equiv \varphi(u)\]
由于有$y = ux$，进一步有：
\begin{align*}
    \dv{y}{x} &= u + x\dv{u}{x}\\
    x\dv{u}{x} &= \varphi(u) - u
\end{align*}
分离变量，最终原齐次微分方程可解：
\begin{align*}
    \frac{\dd{u}}{\varphi(u) - u} &= \frac{\dd{x}}{x}\\
    \int\frac{\dd{u}}{\varphi(u) - u} &= \ln|x| + C
\end{align*}
其中$C$是任意常数，求出上式的不定积分后，再将$u$用$\dfrac{y}{x}$换回，即得到通解.若$\exists u_0$使得$\varphi(u_0)-u_0=0$，则$y=u_0 x$是方程的一个例外解.

\begin{question}
    求解方程：$\displaystyle\dv{y}{x} = \frac{x+y}{x-y}$.
\end{question}

\begin{solution}
    令$y \equiv ux$，分离变量并积分：
    \begin{align*}
        \dv{y}{x} &= \frac{x+y}{x-y}\\
        \int\frac{1-u}{1+u^2}\dd{u} &= \int\frac{\dd{x}}{x}\\
        \arctan u - \frac{1}{2}\ln(1+u^2) &= \ln|x| + C_1
    \end{align*}
    将$u$换回，原方程的通解为：
    \[\sqrt{x^2+y^2} = C \ee^{\arctan\frac{y}{x}}\]
    这个方程是平面上一族以原点为中心的对数螺线.若采取极坐标则写为$r = C\ee^\theta$.
\end{solution}

\begin{question}
    求解方程：$x\dd{y} = (y + \sqrt{x^2+y^2})\dd{x}$.
\end{question}

\begin{solution}
    当$x>0$时，原方程可化为：
    \begin{align*}
        \dv{y}{x} &= \frac{y}{x} + \sqrt{1+\left(\frac{y}{x}\right)^2}\\
        x\dv{u}{x} + u &= u + \sqrt{1+u^2}\\
        \int\frac{\dd{u}}{\sqrt{1+u^2}} &= \int\frac{\dd{x}}{x}\\
        \ln(u+\sqrt{1+u^2}) &= \ln x + C_1
    \end{align*}
    代回$u$得到通解为：
    \[y + \sqrt{x^2+y^2} = Cx^2,\quad C=\ee^{C_1}>0\]

    当$x<0$时，原方程可化为：
    \begin{align*}
        \dv{y}{x} &= \frac{y}{x} + \frac{|x|}{x}\sqrt{1+\left(\frac{y}{x}\right)}\\
        \int\frac{\dd{u}}{\sqrt{1+u^2}} &= -\int\frac{\dd{x}}{x}\\
        \ln|u+\sqrt{1+u^2}| &= -\ln|x| + C_2
    \end{align*}
    去绝对值后，注意到结果与$x>0$时一致，最终原方程的通可以写为：
    \[y = \frac{1}{2}\left(Cx^2 - \frac{1}{C}\right),\quad C>0\]
\end{solution}


\subsection{一阶线性方程}
\begin{definition}
    若一阶微分方程中的$y$和$y'$都是一次的，则称方程为\overtext{线性方程}{linear equation}，并可化为如下标准形式：
    \[\dv{y}{x} + P(x)y = Q(x)\]
    其中，若$Q(x)$恒为$0$，则称之为\textbf{线性齐次方程}；反之则为\textbf{线性非齐次方程}.应当注意，此处的「齐次」与「齐次函数」的「齐次」含义完全不同.
\end{definition}
\subsubsection{线性齐次方程}
对于线性齐次方程，是可分离变量的，注意到$y=0$时有方程的例外解，现考虑$y \neq 0$时有：
\begin{align*}
    \dv{y}{x} + P(x)y &= 0\\
    \int\frac{\dd{y}}{y} &= -\int P(x)\dd{x}\\
    \ln|y| &= -\int P(x)\dd{x} + C_1
\end{align*}
取对数并去绝对值，有$C=\pm\ee^{C_1}$，同时由于$y=0$是例外解，故而$C=0$也成立，整合后有：
\[y = C \ee^{-\int P(x)\dd{x}},\quad C\text{为任意常数}\]


\subsubsection{线性非齐次方程}
对于线性非齐次方程，有两种解法：
\[\dv{y}{x} + P(x)y = Q(x) \neq 0\]
\begin{enumerate}[solution]
    \item 在方程两边同时乘因子$\ee^{\int P(x)\dd{x}}$：
    \begin{align*}
        \dv{y}{x}\cdot\ee^{\int P(x)\dd{x}} + y\cdot P(x)\ee^{\int P(x)\dd{x}} &= Q(x)\cdot\ee^{\int P(x)\dd{x}}\\
        \dv{}{x}\left(y\ee^{\int P(x)\dd{x}}\right) &= Q(x)\cdot\ee^{\int P(x)\dd{x}}\\
        y\ee^{\int P(x)\dd{x}} &= \int Q(x)\ee^{\int P(x)\dd{x}}\dd{x} + C
    \end{align*}
    移项后得到通解为：
    \[y = \underbrace{\left(\int Q(x)\ee^{\int P(x)\dd{x}}\dd{x} + C\right)}_{C(x)}\cdot\ee^{-\int P(x)\dd{x}}\]

    \item \textbf{常数变易法：}对比上一种解法得到的解及线性齐次方程的解，它们的解都是：
    \[y = \text{一个因子} \cdot \ee^{-\int P(x)\dd{x}}\]
    不妨设这个因子为待定的$C(x)$，即令：
    \[y \equiv C(x)\cdot\ee^{-\int P(x)\dd{x}}\]
    代入到原方程中求取：
    \begin{align*}
        \dv{}{x}\left(C(x)\ee^{-\int P(x)\dd{x}}\right) + P(x)\cdot C(x)\ee^{-\int P(x)\dd{x}} &= Q(x)\\
        \dv{C(x)}{x} &= Q(x) \cdot \ee^{-\int P(x)\dd{x}}
    \end{align*}
    两边积分，解得$C(x)$的表达式：
    \[C(x) = \int Q(x)\ee^{\int P(x)\dd{x}}\dd{x} + C,\quad C\text{为任意常数}\]
    将$C(x)$的表达式代回至先前假设的$y$的表达式中，最终得到与第一种解法相同的结果.

    上述方法通过模仿线性齐次方程的解提出试探解，假定原有的$C$不再是一个常数，而是一个变量$C(x)$，代回线性非齐次方程中求解其表达式，再代回试探解进而获得通解.
    
    这样的变换实现了从直接求取$y$转而变成求取$C(x)$间接求取$y$，这就是常数变易法.它适用于在已知线性齐次方程解的情况下快速求取非齐次状态下的解.
\end{enumerate}

\begin{question}
    求解方程：$(2y^2 + y - x)\dd{y} - y\dd{x} = 0$.
\end{question}

\begin{solution}
    注意到$y=0$是一个例外解，现在考虑$y \neq 0$的情况，方程可做如下变形：
    \[\dv{y}{x} = \frac{y}{2y^2+y-x}\]
    这不是一个线性方程，取倒数使之变为线性方程：
    \[\dv{x}{y} + \frac{1}{y}\cdot x = 1+2y\]
    故而这是一个以$y$为自变量$x$为因变量的线性非齐次方程，直接代入其通解的表达式：
    \begin{align*}
        x &= \left(\int(1+2y)\ee^{\int\frac{1}{y}\dd{y}}\dd{y}+C\right)\cdot\ee^{-\int\frac{1}{y}\dd{y}} = \frac{1}{|y|}\left(\int(1+2y)|y|\dd{y} + C\right)\\
        &= \frac{1}{2}y + \frac{2}{3}y^2 + \frac{C}{y},\quad C \text{为任意常数}
    \end{align*}
\end{solution}


\subsubsection{Bernoulli方程}
有一些一阶方程经过适当的代换可以变为一阶线性方程，进而得以求解.以较为常见的Bernoulli方程为例：
\begin{definition}
    称形如下式的微分方程为\textbf{Bernoulli方程}：
    \[\dv{y}{x} + P(x)y = Q(x)y^n,\quad n\in\mathbb{R},n\neq 0,1\]
\end{definition}
注意到对Bernoulli方程有：
\begin{align*}
    \dv{y}{x} + P(x)y &= Q(x)y^n\\
    y^{-n}\dv{y}{x} + P(x)y^{1-n} &= Q(x)\\
    \frac{1}{1-n}\dv{y^{1-n}}{x} + P(x)y^{1-n} &= Q(x)
\end{align*}
令$u \equiv y^{1-n}$有：
\[\dv{u}{x} + (1-n)P(x)u = (1-n)Q(x)\]
由此化为一阶线性非齐次方程.

\begin{question}
    求解方程：$\dv{y}{x} = xy + x^3 y^3$.
\end{question}

\begin{solution}
    注意到对于原方程$y=0$是一个例外解，现考虑$y \neq 0$的情况.

    原方程为Bernoulli方程，设$u \equiv y^{-2}$，将之化为线性方程：
    \[\dv{u}{x} + 2xu = -2x^3\]
    直接代公式求得其通解为$u = 1-x^2+C\ee^{-x^2}$，故而原方程的通解为：
    \[\frac{1}{y^2} = 1-x^2+C\ee^{-x^2}\]
\end{solution}

\begin{extension}{一阶微分方程与向量场}
对于一阶微分方程：
\[\dv{y}{x} = f(x,y) \quad\text{或}\quad (x,y) \mapsto k \mapsto \vect{v}\]
这个微分方程用一个关于点$(x,y)$的函数描述了斜率在二维平面上各点的值，斜率所对应的切线可以看作是一个向量，整个平面上的所有向量则是一个向量场.

也就是说一阶微分方程描述了向量场在平面上的分布，对于不同类型的一阶微分方程，向量场的分布也各有特点.求解微分方程，实际上就是求各点切线与向量场方向一致的曲线的表达式，通解对应一族曲线，特解则对应其中一条曲线.

\begin{itemize}
    \item 若$f(x,y)$是可分离变量的函数，即：
    \[f(x,y) = g(x)h(y)\]
    不难注意到，向量场在各点的值是由$x$和$y$分别控制的，下图中的向量场从每一条水平线或竖直线上看都表现出相同的变化规律：
    \begin{figure}[H]
        \centering
        \includegraphics[width=\textwidth]{differential-equation/separate-variable-equation/fig.pdf}
        \caption{可分离变量的一阶微分方程的解曲线与向量场}
    \end{figure}
    \item 若$f(x,y)$是齐次函数，其向量场则是在过原点的直线上表现出一致的变化规律，即斜率按照对原点距离的$n$次方变化：
    \begin{figure}[H]
        \centering
        \includegraphics[width=\textwidth]{differential-equation/homogeneous-equation/fig.pdf}
        \caption{一阶齐次微分方程的解曲线与向量场}
    \end{figure}
    特别地，对$0$次齐次方程，由于其对距离不敏感，故而每一条过原点的直线上的方向一致；对于$1$次齐次方程，其对距离的一次方和方向敏感，所以关于原点对称的两点的向量恰好是关于$x$轴对称的；对于$2$次齐次方程，由于关于距离的二次方和方向敏感，所以关于原点对称的两点的向量是一致的.
\end{itemize}

\end{extension}
