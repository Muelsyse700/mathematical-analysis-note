\chapter{常微分方程初步}
\begin{definition}
    \overtext{微分方程}{differential equation}指的是联系着一个或多个自变量与函数及其各阶微商的关系式：
    \[F(x,y,y',y'',\cdots,y^{(n)}) = 0\]
    特别对只有单变量的形式称为\overtext{常微分方程}{ordinary differential equation}\textbf{(ODE)}；若方程关于$y,y,\cdots,y^{(n)}$均是一次的，则称之为$n$阶\overtext{线性微分方程}{linear differential equation}.
\end{definition}

\begin{definition}
    设有微分方程$F(x,y,y',y'',\cdots,y^{(n)}) = 0$，若有一函数族：
    \[y = \varphi(x;C_1,C_2,\cdots,C_n)\]
    包含该方程在某一区间上的所有解，其中$C_1,C_2,\cdots,C_n$为任意常数，则称该函数族为该微分方程的\overtext{通解}{general solution}.

    通过在通解中取定任意常数：
    \[C_1 = c_1,\quad C_2=c_2,\quad \cdots, \quad C_n=c_n\]
    所得到的一个具体解，称为该微分方程的\overtext{特解}{particular solution}.
\end{definition}

\section{一阶微分方程}
\subsection{分离变量法}
对于形如$\displaystyle\dv{y}{x}=f(x,y)$的微分方程，若是有：
\[\dv{y}{x} = f(x,y) = g(x)h(y)\]
其中$g,h$都是连续函数且$h(y)$不恒为$0$，则原方程可化为：
\[\frac{1}{h(y)}\dd{y} = g(x)\dd{x}\]
两边积分：
\[\int\frac{1}{h(y)}\dd{y} = \int g(x)\dd{x} \quad\Rightarrow\quad H(y) = G(x) + C\]
上述结果即为原微分方程的通解.特别地，若有$y_0$使得$h(y_0) = 0$，这也是原微分方程的一个解，但是这个解在进行分离变量的时候丢失，从而无法被通解所描述，这样的解称为\textbf{例外解}.

\begin{question}
    求解方程：$(1+x^2)y\dd{y} + \sqrt{1-y^2}\dd{x} = 0$.
\end{question}

\begin{solution}
    当$1-y^2 \neq 0$时，分离变量并积分：
    \begin{align*}
        \int\frac{y}{\sqrt{1-y^2}}\dd{y} &= -\int\frac{1}{1+x^2}\dd{x}\\
        \sqrt{1-y^2} &= \arctan x + C
    \end{align*}
    同时注意到，若有$1-y^2=0$，即$y=\pm1$也满足原微分方程，故而这是例外解.于是这个方程的全部解为：
    \[\sqrt{1-y^2} = \arctan x + C \quad\text{及}\quad y = \pm 1\]
\end{solution}


\subsection{齐次方程}
对于齐次方程，不能直接分离变量，但是经过特定步骤的换元后，就可以通过分离变量法求解.

\begin{definition}
    称$f(x,y)$是$n$次\overtext{齐次函数}{homogeneous function}，若其满足：
    \[f(tx,ty) = t^n f(x,y)\]
    称下述由同次齐次函数$P(x,y),Q(x,y)$构成的满足$f(tx,ty)=f(x,y)$的一阶微分方程为\textbf{齐次方程}：
    \[P(x,y)\dd{x} + Q(x,y)\dd{y} = 0\]
\end{definition}

所谓齐次，就是每一项中所有变量的次数之和一致，它们具有一致的缩放规律，即对输入放大$t$倍，则输出放大$t^n$倍.

从几何上看，如果把输入的$(x,y)$视作一个向量：
\[(x,y) = x \left(1,\frac{y}{x}\right)\]
其中$x$描述的是尺度，$\left(1,\dfrac{y}{x}\right)$则描述了方向，若$f$是$n$次齐次函数，则：
\[f(x,y) = x^n f\left(1,\frac{y}{x}\right) \xlongequal{\varphi(u)\equiv f(1,u)} x^n \varphi(u)\]
上式中$\varphi\left(\frac{y}{x}\right)$实际上就是一个控制单位矢量方向的函数.

如是而言，一个齐次函数可以被看作是：
\[\text{尺度}^n \times \text{仅与方向有关的函数}\]
那么对于$0$次齐次函数，它是一个仅对方向敏感的函数；对于$1$次齐次函数，其是一个对一维尺度和方向敏感的函数，以此类推.

对于齐次方程，由于有$\dfrac{P(x,y)}{Q(x,y)}$是$0$次齐次函数，因此可以将齐次方程化为：
\[\dv{y}{x} = \varphi\left(\frac{y}{x}\right) \equiv \varphi(u)\]
由于有$y = ux$，进一步有：
\begin{align*}
    \dv{y}{x} &= u + x\dv{u}{x}\\
    x\dv{u}{x} &= \varphi(u) - u
\end{align*}
分离变量，最终原齐次微分方程可解：
\begin{align*}
    \frac{\dd{u}}{\varphi(u) - u} &= \frac{\dd{x}}{x}\\
    \int\frac{\dd{u}}{\varphi(u) - u} &= \ln|x| + C
\end{align*}
其中$C$是任意常数，求出上式的不定积分后，再将$u$用$\dfrac{y}{x}$换回，即得到通解.若$\exists u_0$使得$\varphi(u_0)-u_0=0$，则$y=u_0 x$是方程的一个例外解.

\begin{question}
    求解方程：$\displaystyle\dv{y}{x} = \frac{x+y}{x-y}$.
\end{question}

\begin{solution}
    令$y \equiv ux$，分离变量并积分：
    \begin{align*}
        \dv{y}{x} &= \frac{x+y}{x-y}\\
        \int\frac{1-u}{1+u^2}\dd{u} &= \int\frac{\dd{x}}{x}\\
        \arctan u - \frac{1}{2}\ln(1+u^2) &= \ln|x| + C_1
    \end{align*}
    将$u$换回，原方程的通解为：
    \[\sqrt{x^2+y^2} = C \ee^{\arctan\frac{y}{x}}\]
    这个方程是平面上一族以原点为中心的对数螺线.若采取极坐标则写为$r = C\ee^\theta$.
\end{solution}

\begin{question}
    求解方程：$x\dd{y} = (y + \sqrt{x^2+y^2})\dd{x}$.
\end{question}

\begin{solution}
    当$x>0$时，原方程可化为：
    \begin{align*}
        \dv{y}{x} &= \frac{y}{x} + \sqrt{1+\left(\frac{y}{x}\right)^2}\\
        x\dv{u}{x} + u &= u + \sqrt{1+u^2}\\
        \int\frac{\dd{u}}{\sqrt{1+u^2}} &= \int\frac{\dd{x}}{x}\\
        \ln(u+\sqrt{1+u^2}) &= \ln x + C_1
    \end{align*}
    代回$u$得到通解为：
    \[y + \sqrt{x^2+y^2} = Cx^2,\quad C=\ee^{C_1}>0\]

    当$x<0$时，原方程可化为：
    \begin{align*}
        \dv{y}{x} &= \frac{y}{x} + \frac{|x|}{x}\sqrt{1+\left(\frac{y}{x}\right)}\\
        \int\frac{\dd{u}}{\sqrt{1+u^2}} &= -\int\frac{\dd{x}}{x}\\
        \ln|u+\sqrt{1+u^2}| &= -\ln|x| + C_2
    \end{align*}
    去绝对值后，注意到结果与$x>0$时一致，最终原方程的通可以写为：
    \[y = \frac{1}{2}\left(Cx^2 - \frac{1}{C}\right),\quad C>0\]
\end{solution}


\subsection{一阶线性方程}
\begin{definition}
    若一阶微分方程中的$y$和$y'$都是一次的，则称方程为\overtext{线性方程}{linear equation}，并可化为如下标准形式：
    \[\dv{y}{x} + P(x)y = Q(x)\]
    其中，若$Q(x)$恒为$0$，则称之为\textbf{线性齐次方程}；反之则为\textbf{线性非齐次方程}.应当注意，此处的「齐次」与「齐次函数」的「齐次」含义完全不同.
\end{definition}
\subsubsection{线性齐次方程}
对于线性齐次方程，是可分离变量的，注意到$y=0$也满足方程，现考虑$y \neq 0$时有：
\begin{align*}
    \dv{y}{x} + P(x)y &= 0\\
    \int\frac{\dd{y}}{y} &= -\int P(x)\dd{x}\\
    \ln|y| &= -\int P(x)\dd{x} + C_1
\end{align*}
取对数并去绝对值，有$C=\pm\ee^{C_1}$，同时由于$y=0$也是解，故而$C=0$也成立，整合后有：
\[y = C \ee^{-\int P(x)\dd{x}},\quad C\text{为任意常数}\]
这表明一阶线性微分方程不存在例外解，就算存在$y=0$这样的解也总是能被通解所描述.


\subsubsection{线性非齐次方程}
对于线性非齐次方程，有两种解法：
\[\dv{y}{x} + P(x)y = Q(x) \neq 0\]
\begin{enumerate}[solution]
    \item 在方程两边同时乘因子$\ee^{\int P(x)\dd{x}}$：
    \begin{align*}
        \dv{y}{x}\cdot\ee^{\int P(x)\dd{x}} + y\cdot P(x)\ee^{\int P(x)\dd{x}} &= Q(x)\cdot\ee^{\int P(x)\dd{x}}\\
        \dv{}{x}\left(y\ee^{\int P(x)\dd{x}}\right) &= Q(x)\cdot\ee^{\int P(x)\dd{x}}\\
        y\ee^{\int P(x)\dd{x}} &= \int Q(x)\ee^{\int P(x)\dd{x}}\dd{x} + C
    \end{align*}
    移项后得到通解为：
    \[y = \underbrace{\left(\int Q(x)\ee^{\int P(x)\dd{x}}\dd{x} + C\right)}_{C(x)}\cdot\ee^{-\int P(x)\dd{x}}\]

    \item \textbf{常数变易法：}对比上一种解法得到的解及线性齐次方程的解，它们的解都是：
    \[y = \text{一个因子} \cdot \ee^{-\int P(x)\dd{x}}\]
    不妨设这个因子为待定的$C(x)$，即令：
    \[y \equiv C(x)\cdot\ee^{-\int P(x)\dd{x}}\]
    代入到原方程中求取：
    \begin{align*}
        \dv{}{x}\left(C(x)\ee^{-\int P(x)\dd{x}}\right) + P(x)\cdot C(x)\ee^{-\int P(x)\dd{x}} &= Q(x)\\
        \dv{C(x)}{x} &= Q(x) \cdot \ee^{-\int P(x)\dd{x}}
    \end{align*}
    两边积分，解得$C(x)$的表达式：
    \[C(x) = \int Q(x)\ee^{\int P(x)\dd{x}}\dd{x} + C,\quad C\text{为任意常数}\]
    将$C(x)$的表达式代回至先前假设的$y$的表达式中，最终得到与第一种解法相同的结果.

    上述方法通过模仿线性齐次方程的解提出试探解，假定原有的$C$不再是一个常数，而是一个变量$C(x)$，代回线性非齐次方程中求解其表达式，再代回试探解进而获得通解.
    
    这样的变换实现了从直接求取$y$转而变成求取$C(x)$间接求取$y$，这就是常数变易法.它适用于在已知线性齐次方程解的情况下快速求取非齐次状态下的解.
\end{enumerate}

\begin{question}
    求解方程：$(2y^2 + y - x)\dd{y} - y\dd{x} = 0$.
\end{question}

\begin{solution}
    注意到$y=0$是一个例外解，现在考虑$y \neq 0$的情况，方程可做如下变形：
    \[\dv{y}{x} = \frac{y}{2y^2+y-x}\]
    这不是一个线性方程，取倒数使之变为线性方程：
    \[\dv{x}{y} + \frac{1}{y}\cdot x = 1+2y\]
    故而这是一个以$y$为自变量$x$为因变量的线性非齐次方程，直接代入其通解的表达式：
    \begin{align*}
        x &= \left(\int(1+2y)\ee^{\int\frac{1}{y}\dd{y}}\dd{y}+C\right)\cdot\ee^{-\int\frac{1}{y}\dd{y}} = \frac{1}{|y|}\left(\int(1+2y)|y|\dd{y} + C\right)\\
        &= \frac{1}{2}y + \frac{2}{3}y^2 + \frac{C}{y},\quad C \text{为任意常数}
    \end{align*}
\end{solution}


\subsubsection{Bernoulli方程}
有一些一阶方程经过适当的代换可以变为一阶线性方程，进而得以求解.以较为常见的Bernoulli方程为例：
\begin{definition}
    称形如下式的微分方程为\textbf{Bernoulli方程}：
    \[\dv{y}{x} + P(x)y = Q(x)y^n,\quad n\in\mathbb{R},n\neq 0,1\]
\end{definition}
注意到对Bernoulli方程有：
\begin{align*}
    \dv{y}{x} + P(x)y &= Q(x)y^n\\
    y^{-n}\dv{y}{x} + P(x)y^{1-n} &= Q(x)\\
    \frac{1}{1-n}\dv{y^{1-n}}{x} + P(x)y^{1-n} &= Q(x)
\end{align*}
令$u \equiv y^{1-n}$有：
\[\dv{u}{x} + (1-n)P(x)u = (1-n)Q(x)\]
由此化为一阶线性非齐次方程.

\begin{question}
    求解方程：$\dv{y}{x} = xy + x^3 y^3$.
\end{question}

\begin{solution}
    注意到对于原方程$y=0$是一个解，现考虑$y \neq 0$的情况.

    原方程为Bernoulli方程，设$u \equiv y^{-2}$，将之化为线性方程：
    \[\dv{u}{x} + 2xu = -2x^3\]
    直接代公式求得其通解为$u = 1-x^2+C\ee^{-x^2}$，故而原方程的通解为：
    \[\frac{1}{y^2} = 1-x^2+C\ee^{-x^2}\]
\end{solution}


\subsection{可降阶微分方程}
\subsubsection{不显含未知函数的二阶方程}
对于不显含未知函数$y$的二阶方程：
\[f(x,y',y'') = 0\]
引入$p \equiv y'$，则有$\displaystyle y'' = \dv{p}{x}$，原方程变为一阶方程：
\[f(x,p,\dv{p}{x})\]
求出上式的通解$\Phi(x,p,C)$，再将$p$代回，得到一个新的一阶微分方程$\Phi(x,y',C)$，最终求出$y$.

\begin{question}
    求解方程：$xy''+y'=4x$.
\end{question}

\begin{solution}
    令$p \equiv y'$，方程化为：
    \begin{align*}
        xp' + p &= 4x\\
        \dv{(xp)}{x} &= 4x\\
        xp &= 2x^2 + C_1
    \end{align*}
    代回$p$，由此得到一个新的一阶微分方程：
    \begin{align*}
        xy' &= 2x^2 + C_1\\
        \int\dd{y} &= \int \left(2x + \frac{C_1}{x}\right)\dd{x}\\
        y &= x^2 + C_1\ln|x| + C_2
    \end{align*}
\end{solution}

\subsubsection{不显含自变量的二阶方程}
对于不显含自变量的二阶方程：
\[f(y,y',y'') = 0\]
引入$p \equiv y'$，则有：
\[y'' = \dv{p}{x} = \dv{p}{y}\dv{y}{x} = p\dv{p}{y}\]
原方程可降阶为：
\[f\left(y,p,p\dv{p}{y}\right)\]
此时上述方程中$y$变为了自变量，求出上述方程的通解$p = p(y,C)$，于是有新的一阶方程$y' = p(y,C)$，由此求得通解$y$.


\subsection{一阶微分方程的几何意义}
对于一阶微分方程：
\[\dv{y}{x} = f(x,y) \quad\text{或}\quad (x,y) \mapsto k \mapsto \vect{v}\]
这个微分方程用一个关于点$(x,y)$的函数描述了斜率在二维平面上各点的值，斜率所对应的切线可以看作是一个向量，整个平面上的所有向量则是一个向量场.

也就是说一阶微分方程描述了向量场在平面上的分布，对于不同类型的一阶微分方程，向量场的分布也各有特点.求解微分方程，实际上就是求各点切线与向量场方向一致的曲线的表达式，通解对应一族曲线，特解则对应其中一条曲线.
\begin{itemize}
    \item 若$f(x,y)$是可分离变量的函数，即：
    \[f(x,y) = g(x)h(y)\]
    不难注意到，向量场在各点的值是由$x$和$y$分别控制的，下图中的向量场从每一条水平线或竖直线上看都表现出相同的变化规律：
    \begin{figure}[H]
        \centering
        \includegraphics[width=\textwidth]{differential-equation/separate-variable-equation/fig.pdf}
        \caption{可分离变量的一阶微分方程的解曲线与向量场}
    \end{figure}
    \vspace{-2em}
    \item 若$f(x,y)$是齐次函数，其向量场则是在过原点的直线上表现出一致的变化规律，即斜率按照对原点距离的$n$次方变化：
    \begin{figure}[H]
        \centering
        \includegraphics[width=\textwidth]{differential-equation/homogeneous-equation/fig.pdf}
        \caption{一阶齐次微分方程的解曲线与向量场}
    \end{figure}
    \vspace{-2em}
    特别地，对$0$次齐次方程，由于其对距离不敏感，故而每一条过原点的直线上的方向一致；对于$1$次齐次方程，其对距离的一次方和方向敏感，所以关于原点对称的两点的向量恰好是关于$x$轴对称的；对于$2$次齐次方程，由于关于距离的二次方和方向敏感，所以关于原点对称的两点的向量是一致的.
    \item 若是一阶线性微分方程：
    \[\dv{y}{x} + P(x)y = Q(x) \quad\Rightarrow\quad k = -P(x)y + Q(x)\]
    故而对于固定的$x=x_0$，只要$P(x_0) \neq 0$，竖线$x=x_0$上必然有相应的$y = \frac{Q(x_0)}{P(x_0)}$使得斜率为$0$；若在$P(x_0)=0$时有$Q(x_0)=0$，则整条竖线都是对应斜率为$0$的点；若在$P(x_0)=0$时有$Q(x_0)\neq 0$，则整条竖线上都没有对应斜率为$0$的点.这意味着总是能在向量场中画出一条可能存在间断点的斜率为$0$的线.

        \centering
        \includegraphics[width=\textwidth]{differential-equation/linear-equation/fig.pdf}
        \caption{一阶线性微分方程的解曲线与向量场}
    \end{figure}
\end{itemize}


\begin{extension}{动力学与状态空间(相空间)}
    \begin{definition}
        对于一个由常微分方程$f(x,y,y',\cdots,y^{(n)}) = 0$所描述的系统，若这些方程的表达式与自变量$x$无关，则称这个系统是\overtext{自治系统}{autonomous system}.

        在动力学中，自治系统一般指的是不显含时的系统.
    \end{definition}
    对于一阶微分方程，不妨视其为一维上的运动，即：
    \[v \coloneq \dot{x}  = f(t,x) \xlongequal{\text{自治系统}} f(x)\]
    若其为自治系统，则有$x \mapsto v$.不妨认为，整个系统此刻的状态被$x$所唯一地描述，而系统下一刻的状态，可以由对应此刻状态$x$的$v$所预测.

    也就是说，整个由$x$轴所张成的一维空间，包含了系统所有的可能状态.现在有一个自治的一阶微分方程，为这个空间中的每一个状态都赋予了它未来变化的趋势(速度).

    现在考虑二阶微分方程所描述的一维运动：
    \[a \coloneq \ddot{x} = f(t,x,\dot{x}) \xlongequal{\text{自治系统}} f(x,\dot{x}) \quad\text{或}\quad (x,v) \mapsto a\]
    即整个系统此刻的状态仅由两个变量$(x,v)$所描述，由这两个变量所张成的二维空间，包含了整个系统所有的可能状态.

    现在，一个自治的二阶微分方程为这个二维空间中的每一个状态都赋予了未来变化的趋势(加速度).与一维不同的是，这样的趋势包含了两方面的信息，即下一刻$x$如何变化和下一刻$v$如何变化.

    不妨将描述一个系统所有可能状态的空间称为\overtext{状态空间}{state space}\!\!\textbf{(\!\!\overtext{相空间}{phase space}\!\!)}，而相空间上的一个点对应了一个状态.

    而微分方程为相空间中的每一个点都赋予了「动力」，指明了当前状态未来的变化趋势.

    以不忽略空气阻力的单摆系统为例，设摆长为$L$，摆质量为$m$，重力加速度为$g$，空气阻力为$c$.其可由一个二阶微分方程所描述：
    \[\ddot{\theta} = -\frac{g}{L}\sin\theta - \frac{c}{mL^2}\dot{\theta}\]
    显然，这是一个自治系统，其相空间为$(\theta,\dot{\theta})$.
    
    在相空间中用箭头标注二阶微分方程为每一点所指派的变化方向，颜色越深的箭头表明向这个方向的变化程度越高.在相空间中取定一点作为初始状态，不断按时间步进，最终绘制出的轨迹称为\overtext{相轨迹}{phase trajectory}.

    假如在相空间中取所有可能的状态点，这些点都会随时间演化，它们会像流沙一样随向量场流动，形成\overtext{相流}{phase flow}.
    \begin{figure}[H]

\section{二阶线性微分方程}
\begin{definition}
    已知函数$f(x),p(x),q(x)$，求$y=y(x)$满足如下形式的方程：
    \begin{align*}
        y'' + p(x)y' + q(x)y &= f(x) \text{(非齐次)} \\
        y'' + p(x)y' + q(x)y &= 0 \text{(齐次)}
    \end{align*}
    称上述方程为齐次/非齐次\overtext{二阶线性微分方程}{second-order linear differential equation}.
\end{definition}

任何二阶线性微分方程都能够被转化为两个一阶线性微分方程组：
\begin{gather*}
    y'' + p(x)y' + q(x)y = f(x) \quad\text{令}\,\,y_1 \equiv y, y_2 \equiv y'\\
    \Rightarrow\quad \begin{cases}
        y_1' = y' = y_2\\
        y_2' = y'' = f(x) - p(x)y_2 - q(x)y_1
    \end{cases}
\end{gather*}
还可以将方程组写为矩阵形式：
\[\dv{x}\begin{bmatrix}
    y_1 \\ y_2
\end{bmatrix} = \begin{bmatrix}
    0 & 1\\ -q(x) & -p(x)
\end{bmatrix} + \begin{bmatrix}
    0 \\ f(x)
\end{bmatrix}\]

\begin{theorem}
    \label{thm:ExistenceAndUniquenessOfSolutions}
    \textbf{解的存在唯一性}

    若$f(x),p(x),q(x)$在区间$I$上连续，并有$x_0 \in I$，则：
    \[\begin{cases}
        y'' + p(x)y' + q(x)y = f(x)\\
        y(x_0) = \alpha, y'(x_0) = \beta
    \end{cases}\]
    对$\forall \alpha,\beta$，在$x_0$的邻域内存在唯一的解.

    特别地，对齐次二阶线性方程满足初值$y(x_0)=0,y'(x_0)=0$的解恒为$y(x)=0$.
\end{theorem}


\subsection{二阶线性方程解的结构}
\begin{theorem}
    \label{thm:SuperpositionPrincipleForLinearODEs}
    \begin{enumerate}[circ]
        \item 对齐次方程，若$y_1,y_2$是它任意的两个解，则$c_1 y_1 + c_2 y_2$也是原方程的解.
        \item 对非齐次方程，若$y_1,y_2$是其特解，则$y_1 - y_2$是对应齐次方程的特解.
        \item 若$y_0$是非齐次方程的解，则对任意齐次方程的解$y$有$y+y_0$是非齐次方程的解.
    \end{enumerate}
\end{theorem}

上述定理代入至齐次和非齐次方程后相加减易证.上述定理意味着只要能找出齐次方程的通解和一个非齐次方程的特解，就能找到非齐次方程的通解.

\subsubsection{二阶线性齐次方程解的结构}
\begin{definition}
    设$\varphi_1(x),\varphi_2(x)$是区间$I$上的函数，如果$\exists c_1,c_2$不全为$0$，使得对$\forall x \in I$有：
    \[c_1 \varphi_1(x) + c_2 \varphi_2(x) = 0\]
    则称$\varphi_1(x),\varphi_2(x)$是\overtext{线性相关}{linearly dependent}的，反之则为\overtext{线性无关}{linearly independent}的. 
\end{definition}

即对于线性相关的两个函数$\varphi_1(x),\varphi_2(x)$有$c_1 \varphi_1(x) + c_2 \varphi_2(x) = 0$有非零解.两边求导，易知$c_1 \varphi_1'(x) + c_2 \varphi_2'(x) = 0$也有非零解.

联立这两个方程，取其系数行列式：

\begin{definition}
    若$\varphi_1(x),\varphi_2(x)$是$I$上的可导函数，称：
    \[W(x) = \begin{vmatrix}
        \varphi_1(x) & \varphi_2(x) \\
        \varphi_1'(x) & \varphi_2'(x)
    \end{vmatrix}\]
    称$W(x)$为$\varphi_1(x),\varphi_2(x)$的\textbf{Wronski}\overtext{行列式}{determinant}\wiki{https://zh.wikipedia.org/wiki/\%E6\%9C\%97\%E6\%96\%AF\%E5\%9F\%BA\%E8\%A1\%8C\%E5\%88\%97\%E5\%BC\%8F}.
\end{definition}

根据线性方程组有非零解的条件，线性相关的两个函数$\varphi_1(x),\varphi_2(x)$的Wronski行列式等于$0$.但是Wronski行列式为$0$的两个函数不一定线性相关，即也存在线性无关但Wronski行列式为$0$的情况.

但如果两个函数同为线性齐次方程的解，只要Wronski行列式为$0$，就表明二者线性相关：

\begin{theorem}
    \label{thm:WronskianCriterionForLinearDependence}
    设函数$y_1(x),y_2(x)$是线性齐次方程的两个解，则它们在区间$I$上线性相关的充分必要条件是它们在$I$上的Wronski行列式恒为$0$.
\end{theorem}

\begin{proof}
    \textbf{充分性：}由$y_1(x),y_2(x)$线性相关易证.

    \textbf{必要性：}
    不妨取$x=x_0 \in I$，其满足$W(x_0) = 0$，即表明方程组：
    \[\begin{cases}
        c_1 y_1(x) + c_2 y_2(x) = 0\\
        c_1 y_1'(x) + c_2 y_2'(x) = 0\\
    \end{cases}\]
    有非零的解$c_1,c_2$，设有一个新的方程为：
    \[y(x) = c_1 y_1(x) + c_2 y_2(x)\]

    注意到$y_1(x),y_2(x)$是原方程的两个解，根据\cref{thm:SuperpositionPrincipleForLinearODEs}的\circnum{1}，上式中的$y(x)$作为$y_1(x),y_2(x)$的线性组合也是原方程的解.

    代入$x=x_0$到$y(x)$，得到$y(x_0) = 0$；对上述方程两边求导后再代入，得到$y'(x_0)=0$.

    根据\cref{thm:ExistenceAndUniquenessOfSolutions}\textbf{(解的存在唯一性)}，由$y(x)$作为线性齐次方程的解在$x=x_0$时满足的边界条件，故而有：
    \[y(x) = c_1 y_1(x) + c_2 y_2(x) \equiv 0\]
    上式表明具有非零的$c_1,c_2$使得$y_1(x),y_2(x)$的线性组合为$0$，进而其线性相关.
\end{proof}

\begin{theorem}
    \textbf{Liouville公式}

    二阶线性齐次方程的两个解$y_1(x),y_2(x)$的Wronski行列式可以用下述Liouville公式表示：
    \[W(x) = W(x_0)\ee^{-\int_{x_0}^{x}p(t)\dd{t}}\]
    其中$x_0 \in I$.
\end{theorem}

\begin{proof}
    对$W(x)$求导，得到：
    \[\dv{x}W(x) = y_1(x)y_2''(x) - y_1''(x)y_2(x)\]
    由于$y_1(x),y_2(x)$是原方程的两个解，即：
    \[\begin{cases}
        y_1''(x) = - p(x)y_1'(x) - q(x)y_1 \\
        y_2''(x) = - p(x)y_2'(x) - q(x)y_2
    \end{cases}\]
    代入至上式，有：
    \[\dv{x}W(x) = -p(x)(y_1(x)y_2'(x) - y_2(x)y_1'(x)) = -p(x)W(x)\]
    即对一阶线性齐次方程$W'(x) + p(x)W(x) = 0$，其解为：
    \[W(x) = C'\ee^{-\int p(t)\dd{t}}\]
    设$P(t)$为$p(t)$的一个原函数，注意到：
    \[-\int_{x_0}^{x}p(t)\dd{t} = -P(x)+P(x_0) = -\int p(t)\dd{t} + P(x_0)\]
    显然$\ee^{P(x_0)}$是一个常数，于是有：
    \[W(x) = C\ee^{-\int_{x_0}^{x}p(t)\dd{t}},\quad C = C'\ee^{P(x_0)}\]
    取$x = x_0$代入得到$W(x_0) = C$，于是上式进一步写为：
    \[W(x) = W(x_0)\ee^{-\int_{x}^{x_0}p(t)\dd{t}}\]
\end{proof}

\begin{definition}
    若$y_1(x),y_2(x)$是二阶线性齐次微分方程的两个线性无关解，称$(y_1(x),y_2(x))$是该方程的\overtext{基本解组}{fundamental set of solutions}.
\end{definition}

\begin{theorem}
    二阶线性齐次方程$y''(x) + p(x)y' + q(x)y = 0$的基本解组一定存在.
\end{theorem}

\begin{proof}
    $\exists \alpha_1,\alpha_2,\beta_1,\beta_2$使得$\begin{vmatrix}
        \alpha_1 & \alpha_2 \\
        \beta_1 & \beta_2
    \end{vmatrix} \neq 0$，取其作为边界条件.
    
    根据\cref{thm:ExistenceAndUniquenessOfSolutions}\textbf{(解的存在唯一性)}，可知对于在前述边界条件下的二阶线性齐次微分方程：
    \[\begin{cases}
        y''(x) + p(x)y'(x) + q(x)y = 0\\
        y(x_0) = \alpha_1, y'(x_0) = \beta_1
    \end{cases}\]
    一定$\exists y_1(x)$是其唯一解，同理在$y(x_0) = \alpha_2,y'(x_0) = \beta_2$的边界条件下具有唯一解$y_2(x)$.

    对于二阶线性齐次方程的两个解$y_1(x),y_2(x)$，根据Liouville公式有：
    \[W(x) = \begin{vmatrix}
        y_1(x) & y_2(x) \\
        y_1'(x) & y_2'(x)
    \end{vmatrix} = W(x_0)\ee^{-\int_{x_0}^{x}p(t)\dd{t}} = \begin{vmatrix}
        \alpha_1 & \beta_1 \\
        \alpha_2 & \beta_2
    \end{vmatrix}\ee^{-\int_{x_0}^{x}p(t)\dd{t}} \neq 0\]
    即$\exists \alpha_1,\alpha_2,\beta_1,\beta_2$使得$W(x) \neq 0$，根据\cref{thm:WronskianCriterionForLinearDependence}，故而上述$y_1(x),y_2(x)$线性无关，二阶线性齐次方程的基本解组必定存在.
\end{proof}

\begin{theorem}
    若$y_1(x),y_2(x)$是二阶线性齐次方程的一对线性无关解，则原方程的通解可以表示为：
    \[y(x) = c_1 y_1 + c_2 y_2\]
    其中$c_1,c_2$是任意常数.
\end{theorem}

上述定理与\cref{thm:SuperpositionPrincipleForLinearODEs}中\circnum{1}的区别是，前者表明了$y_1(x),y_2(x)$在线性无关的条件下的线性组合具有通解性，后者没有限定其线性无关，它们的线性组合不具有通解性.

\begin{proof}
    根据\cref{thm:SuperpositionPrincipleForLinearODEs}的\circnum{1}，对于二阶线性齐次方程任意的一个解$y(x)$，其可以分解为：
    \[c_1 y_1(x) + c_2 y_2(x) = y(x)\]
    沿用上一个定理证明所设的变量，取$x=x_0$代入上式，而后对上式求导后再代入得到：
    \[\begin{cases}
        \alpha_1 c_1 + \alpha_2 c_2 = y(x_0) \\
        \beta_1 c_1 + \beta_2 c_2 = y'(x_0)
    \end{cases} \Leftrightarrow\quad \begin{bmatrix}
        \alpha_1 & \alpha_2 \\
        \beta_1 & \beta_2
    \end{bmatrix} \begin{bmatrix}
        c_1 \\ c_2
    \end{bmatrix} = \begin{bmatrix}
        y(x_0) \\ y'(x_0)
    \end{bmatrix}\]
    上述方程的解$c_1,c_2$存在的条件是$\begin{vmatrix}
        \alpha_1 & \beta_1 \\
        \alpha_2 & \beta_2
    \end{vmatrix} \neq 0$，故而只有在$y_1(x),y_2(x)$线性无关时，它们的线性组合才是原方程的通解.
\end{proof}

在上述诸多定理推导的过程中得知，对于一个二阶线性齐次方程，其必然存在基本解组，若有基本解组，就能基于线性组合得到原方程的通解.

那么，求解基本解组成为了当前的主要矛盾.不妨先假设基本解组$(y_1(x),y_2(x))$中，$y_1(x)$是已知的一个非零解.

必定$\exists x_0$使得$W(x_0) = 1$，于是有：
\[W(x) = y_1(x)y_2'(x) - y_1'(x)y_2(x) = \ee^{-\int_{x_0}^{x}p(t)\dd{t}}\]
两边同除$y_1^2(x)$，构造除法的导数：
\begin{align*}
    \frac{y_1(x)y_2'(x) - y_1'(x)y_2(x)}{y_1^2(x)} &= \frac{1}{y_1^2(x)}\ee^{-\int_{x_0}^{x}p(t)\dd{t}}\\
    \dv{x}\left(\frac{y_2(x)}{y_1(x)}\right) &= \frac{1}{y_1^2(x)}\ee^{-\int_{x_0}^{x}p(t)\dd{t}}
\end{align*}
两边积分：
\[y_2(x) = y_1(x)\int\frac{1}{y_1^2(x)}\ee^{-\int_{x_0}^{x}p(t)\dd{t}}\dd{x}\]
上式表明，只要能得到二阶线性齐次方程的一个非零解$y_1(x)$，就能得到第二个解，进而得到通解.

而对于求取第一个解的方法，则要具体问题具体分析.

\begin{question}
    求方程$xy'' - y' = 0$的通解.
\end{question}

\begin{solution}
    原方程可化为：
    \[y'' - \frac{1}{x}y' = 0\]
    记$p(x) = \dfrac{1}{x}$.显然$y_1(x)= 1$是原方程的一个非零解，于是有另一个解：
    \[y_2(x) = \int \exp(-\int_{x_0}^{x}\frac{1}{t}\dd{t})\dd{x} = C\int \left|\frac{x}{x_0}\right|\dd{x} = C_2 x^2\]
    对于$y_1(x),y_2(x)$有$W(x) \neq 0$，则通解为：
    \[y(x) = C_1 + C_2(x)\]
\end{solution}


\subsubsection{二阶线性非齐次方程解的结构}
若$y_1(x),y_2(x)$是二阶线性齐次方程的线性无关解，那么有$c_1 y_1(x) + c_2 y_2(x)$是该方程的通解.

又根据\cref{thm:SuperpositionPrincipleForLinearODEs}的\circnum{3}，若已知$y_0(x)$是相应非齐次方程的解，则：
\[y(x) = y_0(x) + c_1 y_1(x) + c_2 y_2(x)\]
根据$c_1,c_2$的任意性，$y(x)$是二阶线性非齐次方程的通解.

也就是说，只要有非齐次方程的一个特解，和齐次方程的一个非零特解，那么就能求出非齐次方程的通解.
