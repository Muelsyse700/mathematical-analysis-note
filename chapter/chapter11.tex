\chapter{曲线积分和曲面积分}
在一元函数的积分中，通过计算$f(x)$的原函数$F(x)$满足：
\[\dd{F(x)} = f(x)\dd{x}\]
进一步基于Newton-Leibniz公式通过区域的边界计算区域内部导数积分的值.而在计算重积分的过程中，往往是使用各种方法将其转换为累次积分，再使用Newton-Leibniz公式.

是否存在、能否求取多元函数的原函数$F(x,y)$使得：
\[\dd{F(x,y)} = F'_x(x,y)\dd{x} + F'_y(x,y)\dd{y} = \nabla F \cdot \dd{\vb{r}}\]
进而推广出高维空间的Newton-Leibniz公式，将区域内部导数的积分与区域边界上的函数值相联系？这就是Gauss-Stokes公式.


\section{数量场在曲线上的积分}
要想求取原函数$F(x,y)$，就要尝试作积分：
\[\iint_D \dd{F(x,y)} = \int_L \nabla F \cdot \dd{\vb{r}}\]
显然，后者是一个向量场上的积分，在解决这个问题之前，首先要解决数量场上的积分.


\subsection{基本概念}
\begin{definition}
    \label{def:LineIntegral1}
    设$\mathbb{R}^3$中有一条光滑曲线$L$，数量场$\varphi(x,y,z)$定义在包含$L$的区域$D$上且连续.作$L$的任意分割$M_0,\cdots,M_n$，每一段弧$\wideparen{M_{i-1}M_i}$上任取一点$N_i(\xi_i,\eta_i,\zeta_i)$，记弧长为$\Delta s_i$，取$\lambda = \max\{|\Delta s_i|,i=1,\cdots,n\}$，如果下列求和的极限存在：
    \[\lim_{\lambda\to0}\sum_{i=1}^{n}\varphi(\xi_i,\eta_i,\zeta_i)\Delta s_i \eqcolon \int_L\varphi(x,y,z)\dd{s}\]
    则称函数或数量场$\varphi(x,y,z)$在曲线$L$上可积.相应的极限值称为\overtext{第一型曲线积分}{line integral of the first kind}或\overtext{数量场在曲线上的积分}{line integral of a scalar field}.
\end{definition}

\subsection{数量场在曲线上积分的计算}
对于光滑曲线$L$的参数方程：
\[L:\vb{r}(t) = x(t)\vb{i} + y(t)\vb{j} + z(t)\vb{k},\quad t\in[\alpha,\beta]\]
在参数区间上作对应分割点$M_1,\cdots,M_n$的分割：
\[T:\alpha = t_0 < t_1 < \cdots < t_n = \beta\]
根据积分中值定理和切向量的计算公式，有：
\[\Delta s_i = \int_{t_{i-1}}^{t_i}|\vb{r}'(t)|\dd{t} = |\vb{r}'(\theta_i)|\Delta t_i,\quad \theta_i \in [t_{i-1},t_i]\]
设$(\xi_i,\eta_i,\zeta_i) = (x(\tau_i),y(\tau_i),z(\tau_i))$，其中$\tau\in[t_{i-1},t_i]$，故而\cref{def:LineIntegral1}中的求和有：
\[\sum_{i=1}^{n}\varphi(x(\tau_i),y(\tau_i),z(\tau_i))\Delta s_i = \sum_{i=1}^{n}\varphi(x(\tau_i),y(\tau_i),z(\tau_i))|\vb{r}'(\theta_i)|\Delta t_i\]
对上式右边进行修正，使其满足Riemann和的形式：
\[\text{右边} = \sum_{i=1}^{n}\varphi(x(\tau_i),y(\tau_i),z(\tau_i))|\vb{r}'(\tau_i)|\Delta t_i + \sum_{i=1}^{n}\varphi(x(\tau_i),y(\tau_i),z(\tau_i))[|\vb{r}'(\theta_i)|-|\vb{r}'(\tau_i)|]\Delta t_i\]
由于$L$是光滑曲线，故而$\vb{r}'(t)$连续，进而通过$\varphi(x(\tau_i),y(\tau_i),z(\tau_i))|\vb{r}'(\tau_i)|$连续推出其可积.

根据可积的定义，对$\forall\varepsilon>0$，$\exists\delta_1>0$，在$\left\|T\right\|<\delta_1$时有：
\[\left|\sum_{i=1}^{n}\varphi(x(\tau_i),y(\tau_i),z(\tau_i))|\vb{r}'(\tau_i)|\Delta t_i - \int_{\alpha}^{\beta}\varphi(x(t),y(t),z(t))|\vb{r}'(t)|\dd{t}\right| < \varepsilon\]
在闭区间$[\alpha,\beta]$上，$\varphi(x(t),y(t),z(t))$因其连续而有界，即$|\varphi|\leqslant M$.而$|\vb{r}'(t)|$因其连续而一致连续，即$\exists\delta_2$，对$\forall t,t'\in[\alpha,\beta]$，在$|t-t'|<\delta_2$时有：
\[||\vb{r}'(t)|-|\vb{r}'(t')||<\varepsilon\]
进一步当$\left\|T\right\|<\delta_2$时，对$\forall\theta_i,\tau_i\in[t_{i-1},t_i]$有$|\theta_i-\tau_i|\leqslant\Delta t_i \leqslant\left\|T\right\|\leqslant\delta_2$，从而有：
\[\left|\sum_{i=1}^{n}\varphi(x(\tau_i),y(\tau_i),z(\tau_i))[|\vb{r}'(\theta_i)|-|\vb{r}'(\tau_i)|]\Delta t_i\right| < M(\beta-\alpha)\varepsilon\]
当$\left\|T\right\| < \delta = \min\{\delta_1,\delta_2\}$时：
\[\left|\sum_{i=1}^{n}\varphi(x(\tau_i),y(\tau_i),z(\tau_i))|\vb{r}'(\tau_i)|\Delta s_i - \int_{\alpha}^{\beta}\varphi(x(t),y(t),z(t))|\vb{r}'(t)|\dd{t}\right| < \varepsilon + M(\beta-\alpha)\varepsilon\]
由此得到：
\begin{theorem}
    设$\varphi(x,y,z)$在光滑曲线$L$上连续，则其在$L$上可积：
    \begin{align*}
        \int_L \varphi(x,y,z)\dd{s} &= \int_{\alpha}^{\beta}\varphi(x(t),y(t),z(t))|\vb{r}'(t)|\dd{t} \\
        &=\int_{\alpha}^{\beta}\varphi(x(t),y(t),z(t))\sqrt{x'^2(t)+y'^2(t)+z'^2(t)}\dd{t}
    \end{align*}
\end{theorem}


\section{数量场在曲面上的积分}
\subsection{曲面的面积}
设有一个从$O'uv$平面上有界区域$D$到$Oxyz$空间中一张光滑参数曲面$S$的映射：
\[(u,v)\mapsto(x,y,z):\vb{r}(u,v) = x(u,v)\vb{i} + y(u,v)\vb{j} + z(u,v)\vb{k},\quad (u,v)\in D \subset\mathbb{R}^2\]
在区域$D$上作矩形分割，得到一个小矩形区域$D_{ij}$到$S$上一个子曲面$S_{ij}$的映射：
\[D_{ij} = \{(u,v)\,|\,(u,v)\in[u_i,u_{i+1}]\times[v_i,v_{i+1}]\}\quad \to \quad S_{ij}\]
当$\Delta u_i,\Delta v_i$都很小时，$S_{ij}$近似为一个平行四边形，其一对邻边的向量可表示为：
\begin{gather*}
    \vb{r}(u_{i+1},v_j) - \vb{r}(u_i,v_j) = \vb{r}'_u(u_i,v_j)\Delta u_i + o(\Delta u_i)\\
    \vb{r}(u_i,v_{j+1}) - \vb{r}(u_i,v_j) = \vb{r}'_v(u_i,v_j)\Delta v_j + o(\Delta v_j)\\
    \Rightarrow\quad \sigma(S_{ij}) \approx |\vb{r}'_u(u_i,v_j) \times \vb{r}'_v(u_i,v_j)|\Delta u_i \Delta v_j
\end{gather*}
积分得到曲面$S$的面积：
\[\sigma(S) = \iint_D |\vb{r}'_u(u_i,v_j) \times \vb{r}'_v(u_i,v_j)|\dd{u}\dd{v}\]

\begin{mistake}
    这样的映射过程令人想到重积分换元时的映射，变换因子就是相应的Jacobi行列式.但是换元映射是在同维度下进行的，而这里是把$\mathbb{R}^2$的网格映射到$\mathbb{R}^3$上的网格，尽管二者都是面，但是维度不同.
\end{mistake}

引入在微分几何中常用的记号：
\begin{gather*}
    E \coloneq {\vb{r'}_u}^2,\quad G \coloneq {\vb{r'}_v}^2,\quad F = \vb{r'}_u \cdot \vb{r'}_v \\
    \dd{S} \coloneq |\vb{r}'_u(u_i,v_j) \times \vb{r}'_v(u_i,v_j)|\dd{u}\dd{v} = \sqrt{EG-F^2}\dd{u}\dd{v}
\end{gather*}
称$\dd{S}$为曲面的\textbf{面积元素}.

有以下几类特例：
\begin{itemize}
    \item $F = \vb{r}'_u \cdot \vb{r}'_v = 0$，意味着$u$曲线和$v$曲线在$S$上的网格是处处垂直的.
    \item 当$S \in \mathbb{R}^2$时，类似重积分换元，变换因子就是相应的Jacobi行列式，即：
    \[|\vb{r}'_u\times\vb{r}'_v| = \left|\pdv{(x,y)}{(u,v)}\right| \quad\Rightarrow\quad \dd{\sigma} = \left|\pdv{(x,y)}{(u,v)}\right|\dd{u}\dd{v}\]
    \item 若曲面$S$由二元函数$z = f(x,y)$给出，曲面面积公式可写为：
    \[\sigma(S) = \iint_D\sqrt{1+{f'_x}^2 + {f'_y}^2}\dd{x}\dd{y}\]
\end{itemize}


\subsection{数量场在曲面上积分的计算}
\begin{definition}
    设$\mathbb{R}^3$中有一张光滑曲面$S$，数量场$\varphi(x,y,z)$定义在包含$S$的区域内且连续.将$S$分割为$n$块子曲面$S_i$，记其面积为$\sigma(S_i)$.在$S_i$上任取一点$M_i(\xi_i,\eta_i,\zeta_i)$，如果下列极限存在：
    \[\lim_{\lambda\to0}\sum_{i=1}^{n}\varphi(\xi_i,\eta_i,\zeta_i)\sigma(S_i) \eqcolon \iint_S\varphi(x,y,z)\dd{S}\]
    则称$\varphi(x,y,z)$在曲面$S$上可积，相应的极限值称为\overtext{第一型曲面积分}{surface integral of the first kind}.
\end{definition}

\begin{theorem}
    设$\varphi(x,y,z)$在光滑曲面$S$上连续，则其在$S$上可积：
    \begin{align*}
        \iint_S \varphi(x,y,z)\dd{S} &= \iint_D \varphi(x(u,v),y(u,v),z(u,v))|\vb{r'}_u\times\vb{r'}_v|\dd{u}\dd{v} \\
        &= \iint_D \varphi(x(u,v),y(u,v),z(u,v))\sqrt{EG-F^2}\dd{u}\dd{v}
    \end{align*}
\end{theorem}

上述定理的证明过程类似于二重积分和曲线积分计算方法证明过程的结合.


\section{向量场在曲线上的积分}
\subsection{曲线的定向}
对于一个向量场，例如重力场，沿着重力方向移动其做正功，逆向移动则做负功.从做功的数值上看，只有正负号的差别，但是在物理意义上大有不同.在物理上可以根据模型本身规定正负，但是在数学上就要做一个统一的约定.

\begin{definition}
    设$L$是连接空间两点$A,B$的曲线，规定其中一点为起点，就规定了$L$的方向.具有方向定义的曲线称为\overtext{定向曲线}{oriented curve}，以$A$为起点的记作$L_{AB}$，反之则为$L_{BA}$.对于以参数方程表示的$L$：
    \[L:\vb{r}(t) = x(t)\vb{i} + y(t)\vb{j} + z(t)\vb{k},\quad \alpha\leqslant t \leqslant\beta\]
    当$t$由$\alpha$向$\beta$递增时，$L$上对应点由$A$向$B$运动，则称$t$是$L_{AB}$的\overtext{正向参数}{positive parameter}.

    对于逐段光滑曲线，可以逐段选取正向参数，得到分段表示.
    
    对于平面上的简单闭曲线，则习惯上称其逆时针方向为正方向，或者说$L$内部区域在其正方向左边.
\end{definition}

对于参数曲线，其切向量总是指向参数增加的方向.因此指定单位切向量为：
\[\begin{cases}
    \vb{\tau} = \dfrac{\vb{r'}(t)}{|\vb{r'}(t)|} \quad&\Leftrightarrow\quad \text{正方向} \\[1em]
    \vb{\tau} = -\dfrac{\vb{r'}(t)}{|\vb{r'}(t)|} \quad&\Leftrightarrow\quad \text{负方向} \\
\end{cases}\]


\subsection{向量场在曲线上积分的计算}
\begin{definition}
    设有空间区域$D$中的向量场$\vb{v} = P(x,y,z)\vb{i} + Q(x,y,z)\vb{j} + R(x,y,z)\vb{k}$，在$D$内的定向曲线$L_{AB}$上沿其方向作分割$T:A = M_0,M_1,\cdots,M_n=B$，所截取得的有向弧段可表示为：
    \[\Delta\vb{r}_i = \va{M_{i-1}M_i} = \Delta x_i \vb{i} + \Delta y_i \vb{j} + \Delta z_i \vb{k}\]
    在弧段$\wideparen{M_{i-1}M_i}$上任取一点$N_i(\xi_i,\eta_i,\zeta_i)$，若下列和式的极限存在：
    \[\lim_{\left\|T\right\|\to0}\sum_{i=1}^{n}\vb{v}(\xi_i,\eta_i,\zeta_i)\cdot\Delta\vb{r}_i \eqcolon \int_{L_{AB}}\vb{v}\cdot\dd{\vb{r}}\]
    则称极限值为向量场$\vb{v}$沿路径$L_{AB}$可积，称$L_{AB}$为\overtext{积分路径}{integration path}，极限值称为\overtext{第二型曲线积分}{line integral of the second kind}.

    特别地，若$L$是封闭曲线，积分也称为向量场$\vb{v}$沿环路$L$的\overtext{环量}{circulation}，记作：
    \[\oint_L \vb{v}\cdot\dd{\vb{r}}\]
\end{definition}

对$L_{AB}$的参数区间作分割对应曲线上点的分割，令分割趋于$0$，根据Lagrange中值定理，$\exists \lambda_i,\mu_i,\nu_i \in [t_{i-1},t_i]$使得：
\[\Delta\vb{r}_i = \Delta x_i\vb{i} + \Delta y_i\vb{j} + \Delta z_i\vb{k} = x'(\lambda_i)\Delta t_i \vb{i} + y'(\mu_i)\Delta t_i \vb{j} + z'(\nu_i)\Delta t_i \vb{k}\]
在$[t_{i-1},t_i]$上再任取一点$\tau_i$使得$(x(\tau_i),y(\tau_i),z(\tau_i)) = (\xi_i,\eta_i,\zeta_i)$，由此将定义中的求和转换为一个类Riemann和：
\begin{align*}
    \sum_{i=1}^{n}\vb{v}(\xi_i,\eta_i,\zeta_i)\cdot\Delta\vb{r}_i =& \sum_{i=1}^{n}P(x(\tau_i),y(\tau_i),z(\tau_i))x'(\lambda_i)\Delta t_i \\
    &+ \sum_{i=1}^{n}Q(x(\tau_i),y(\tau_i),z(\tau_i))y'(\mu_i)\Delta t_i \\
    &+ \sum_{i=1}^{n}R(x(\tau_i),y(\tau_i),z(\tau_i))z'(\nu_i)\Delta t_i
\end{align*}
仿照第一型曲线积分的处理方法，将上式转换为三个Riemann和，并且修正项极限为$0$.最终得到：

\begin{theorem}
    设向量场$\vb{v}(x,y,z)$在含有光滑有向曲线$L_{AB}$的$D$上连续，则其在$L_{AB}$上可积：
    \[\int_{L_{AB}}\vb{v}\cdot\dd{\vb{r}} = \int_{\alpha}^{\beta}\vb{v}(\vb{r}(t))\cdot\vb{r}'(t)\dd{t} = \int_{\alpha}^{\beta}[Px'(t) + Qy'(t) + Rz'(t)]\dd{t}\]
\end{theorem}
向量场在曲线上的积分具有以下性质：
\begin{enumerate}[property]
    \item \textbf{线性性}：若$\vb{v} = c_1\vb{v_1} + c_2\vb{v_2}$，则有：
    \[\int_L\vb{v}\cdot\dd{\vb{r}} = c_1\int_L\vb{v_1}\cdot\dd{\vb{r}} + c_2\int_L\vb{v_2}\cdot\dd{\vb{r}}\]
    \item \textbf{积分曲线的可加性}：若$L_{AC}$是由$L_{AB}$和$L_{BC}$连接而成，则有：
    \[\int_{L_{AC}}\vb{v}\cdot\dd{\vb{r}} = \int_{L_{AB}}\vb{v}\cdot\dd{\vb{r}} + \int_{L_{BC}}\vb{v}\cdot\dd{\vb{r}}\]
    \item \textbf{积分的方向性}：
    \[\int_{L_{AB}}\vb{v}\cdot\dd{\vb{r}} = -\int_{L_{BA}}\vb{v}\cdot\dd{\vb{r}}\]
\end{enumerate}

\begin{question}
    设有引力场$\vb{F}=-k\dfrac{mM}{r^3}\vb{r}$，求质点$m$从一点$A$到另一点$B$运动时引力所做的功.
\end{question}

\begin{solution}
    \begin{gather*}
        W = \int_{L_{AB}}\vb{F}\cdot\dd{\vb{r}} = -kmM\int_{L_{AB}}\frac{1}{r^3}\vb{r}\cdot\dd{\vb{r}}\\
        \vb{r}\cdot\dd{\vb{r}} = (xx'+yy'+zz')\dd{t} = \frac{1}{2}\dv{t}[x^2(t)+y^2(t)+z^2(t)]\dd{t} = \frac{1}{2}\dv{r^2}{t}\dd{t} = r\dd{r}\\
        \Rightarrow\quad W = -kmM\int_{r_A}^{r_B}\frac{1}{r^2}\dd{r} = kmM\left(\frac{1}{r_B}-\frac{1}{r_A}\right)
    \end{gather*}
    上述推导表明，在重力场中重力做功只与始末位置有关，与具体路径无关.
\end{solution}


\subsection{Green定理}
向量场沿路径的积分有的与路径有关，有的与路径无关，如何判别这两种向量场？

设有一个二维光滑向量场$\vb{v} = P(x,y)\vb{i} + Q(x,y)\vb{j}$，有两条不同的连接$A$到$B$的路径$L_{AB}$和$\tilde{L}_{AB}$.则$L_{AB}$和$\tilde{L}_{BA}$构成一条封闭有向环路，若对于任意不自相交的封闭环路$L$，其路径积分为$0$，则该向量场的路径积分与路径无关.

\begin{theorem}
    \textbf{Green定理}
    \label{thm:GreenTheorem}

    设$D$是有限条逐段光滑的封闭曲线$L$围成的平面闭区域，即$L=\partial D$，$\vb{v}=P(x,y)\vb{i}+Q(x,y)\vb{j}$是$D$上的光滑向量场，则：
    \[\oint_{\partial D} P\dd{x} + Q\dd{y} = \iint_D\left(\pdv{Q}{x}-\pdv{P}{y}\right)\dd{x}\dd{y}\]
\end{theorem}

\begin{proof}
    \circnum{1}
    设有曲边矩形区域$D_y$由两条光滑曲线$L_1,L_2$和两条平行于$y$轴的直线$L_2,L_4$围成：
    \[\begin{cases}
        L_1:y = \varphi_1(x) \\ L_3:y = \varphi_2(x)
    \end{cases},x\in[a,b]\qquad\qquad \begin{cases}
        L_2:x = b \\ L_4:x = a
    \end{cases}\]
    设$D_y$上有向量场$\vb{v}_x = P(x,y)\vb{i}$.
    在$L_1$和$L_3$上$\vb{v}_x$分别满足：
    \[\int_{L_1}P\dd{x} = \int_{a}^{b}P(x,\varphi_1(x))\dd{x}\qquad \int_{L_3}P\dd{x} = \int_{b}^{a}P(x,\varphi_2(x))\dd{x}\]
    约定从$|\vb{r}|$最小的点逆时针出发，先经过较小的$\varphi_1(x)$，再经过较大的$\varphi_2(x)$：
    \begin{align*}
        \int_{L_1}P\dd{x} - \int_{L_3}P\dd{x} &= \int_{a}^{b}P(x,\varphi_1(x))\dd{x} - \int_{a}^{b}P(x,\varphi_2(x))\dd{x} \\
        &= \int_{a}^{b}[P(x,\varphi_1(x)) - P(x,\varphi_2(x))]\dd{x} \\
        &= -\int_{a}^{b}\dd{x}\int_{\varphi_1(x)}^{\varphi_2(x)}\pdv{P(x,y)}{y}\dd{y} \\
        &= -\iint_{D_y}\pdv{P(x,y)}{y}\dd{x}\dd{y}
    \end{align*}
    在$L_2$和$L_4$上$P$由于$\dd{x}=0$因而其贡献为$0$，故而$\vb{v}_x$在$D_y$上满足Green公式：
    \[\oint_{\partial D_y}\vb{v}_x\cdot\dd{\vb{r}} = -\iint_{D_y}\pdv{P}{y}\dd{x}\dd{y}\]
    \textit{注意：负号来源于Newton-Leibniz公式对下限小于上限的要求.}

    设有另一种曲边矩形区域$D_x = \{(x,y)\,|\,\psi_1(y)\leqslant x \leqslant \psi_2(y),c\leqslant y \leqslant d\}$，其上有向量场$\vb{v}_y = Q(x,y)\vb{j}$，类似关于$D_y$型区域的推导，在$D_x$上有：
    \[\oint_{\partial D_x}\vb{v}_y\cdot\dd{\vb{r}} = \iint_{D_x}\pdv{Q}{x}\dd{x}\dd{y}\]
    \textit{注意：此处没有负号，是因为约定从$|\vb{r}|$最小的点出发，在逆时针方向上先经过较大的$\psi_2(y)$，再经过较小的$\psi_1(y)$，天然满足Newton-Leibniz公式的顺序.}

    \circnum{2}
    设区域$D$能够分解为无公共内点，边界为逐段光滑曲线的若干小块：
    \[D = \bigcup_{i=1}^n D_i\]
    $\partial D_i$均取逆时针方向，如果Green公式在每个$D_i$上成立，则对整个$D$也成立.因为所有公共边界上的积分被相邻小块上不同方向的积分相抵消，则有：
    \begin{align*}
        \oint_{\partial D} P\dd{x} + Q\dd{y} &= \sum_i \oint_{\partial D_i}P\dd{x} + Q\dd{y} = \sum_i\iint_{D_i}\left(\pdv{Q}{x}-\pdv{P}{y}\right)\dd{x}\dd{y} \\
        &= \iint_D\left(\pdv{Q}{x}-\pdv{P}{y}\right)\dd{x}\dd{y}
    \end{align*}
    根据以上推导，若$D$可以分割为若干$D_y$型区域，对于向量场$\vb{v}_x$有：
    \[\oint_{\partial D}\vb{v}_x\cdot\dd{\vb{r}} = \iint_{D}\left(-\pdv{P}{y}\right)\dd{x}\dd{y}\]
    若$D$同时也可以分割为若干$D_x$型区域，对于向量场$\vb{v}_y$有：
    \[\oint_{\partial D}\vb{v}_y\cdot\dd{\vb{r}} = \iint_{D}\left(\pdv{Q}{x}\right)\dd{x}\dd{y}\]
    注意到$\vb{v} = P(x,y)\vb{i}+Q(x,y)\vb{j} = \vb{v}_x + \vb{v}_y$，则将以上二式相加则有：
    \[\oint_{\partial D}\vb{v}\cdot\dd{\vb{r}} = \oint_{\partial D}(\vb{v}_x+\vb{v}_y)\cdot\dd{\vb{r}} = \iint_D\left(\pdv{Q}{x}-\pdv{P}{y}\right)\dd{x}\dd{y}\]
    由此得证.
\end{proof}
上述证明依然存在一些缺陷，没有给出Green定理适用区域的完整定义.

在求取向量场在一些难以用参数方程表示的曲线上的积分时，可以将曲线连成封闭曲线，转而使用Green定理求解，而后再减掉新加入连线上的积分.

通过构造特殊的向量场，Green定理可用于求取面积：
\begin{theorem}
    设$D$是满足Green定理条件的区域，且具有分段光滑的边界，则有：
    \[\sigma(D) = \frac{1}{2}\oint_{\partial D}(-y\dd{x}+x\dd{y}) = \oint_{\partial D}(-y)\dd{x} = \oint_{\partial D}x\dd{y}\]
\end{theorem}

对于积分与路径无关的向量场，有如下结论：
\begin{theorem}
    设$D$是平面上的单连通区域，$\vb{v}$是$D$上的光滑向量场，则以下三个命题相互等价：
    \begin{enumerate}[circ]
        \item 向量场$\vb{v}$在区域$D$中的曲线积分与路径无关，记为：
        \[\oint\vb{v}\cdot\dd{\vb{r}} = 0 \quad\text{或}\quad \int_{L_{AB}}\vb{v}\cdot\dd{\vb{r}} = \int_{A}^{B}\vb{v}\cdot\dd{\vb{r}}\]
        \item 向量场$\vb{v}$是一个函数$\varphi(x,y)$的梯度场：
        \[\vb{v} = \grad\varphi(x,y) = \nabla\varphi(x,y)\]
        且这个函数$\varphi(x,y)$的非常数项是唯一的.
        \item 向量场$\vb{v}$的两个分量满足：
        \[\pdv{Q}{x}-\pdv{P}{y} = 0\]
    \end{enumerate}
\end{theorem}

\begin{proof}
    \circnum{1}$\Rightarrow$\circnum{2}：取$D$内一个固定点$(x_0,y_0)$和一个动点$(x,y)$，由于积分与路径无关，于是下列积分定义了一个函数映射：
    \[\varphi:(x,y)\mapsto\int_{(x_0,y_0)}^{(x,y)}\vb{v}\cdot\dd{\vb{r}} = \int_{(x_0,y_0)}^{(x_y)}P\dd{x}+Q\dd{y}\]
    若动点位于$(x_0+\Delta x,y_0)$，则$\dd{y}=0$，进而有：
    \[\frac{\varphi(x_0+\Delta x,y_0) - \varphi(x_0,y_0)}{\Delta x} = \frac{1}{\Delta x}\int_{(x_0,y_0)}^{(x_0+\Delta x,y_0)}P\dd{x}\]
    令$\Delta x \to 0$得到$\displaystyle\pdv{\varphi}{x} = P$.同理可得$\displaystyle\pdv{\varphi}{y} = Q$，即：
    \[\vb{v} = \nabla\varphi(x,y) = \pdv{\varphi}{x}\dd{x}+\pdv{\varphi}{y}\dd{y}\]

    \circnum{2}$\Rightarrow$\circnum{3}：由于$\vb{v}$是光滑向量场，因而对$\varphi(x,y)$的混合偏导次序可以交换：
    \[\pdv{x}\pdv{\varphi}{y} - \pdv{y}\pdv{\varphi}{x} = 0\]

    \circnum{3}$\Rightarrow$\circnum{1}：根据Green定理计算显然可得.
\end{proof}

\begin{theorem}
    设$D$是平面上的单连通区域，$\vb{v}$是$D$上的光滑向量场，如果$\vb{v}$在$D$内的积分与路径无关，则对于区域$D$内的任意两点$A,B$有：
    \[\int_{A}^{B}\vb{v}\cdot\dd{\vb{r}} = \varphi(B) - \varphi(A),\quad \nabla\varphi = \vb{v}\]
    \textit{这个定理类似于单变量的Newton-Leibniz公式.}
\end{theorem}

\begin{proof}
    任取一条连接$A,B$的曲线$L_{AB}$：
    \[L_{AB}:x = x(t), y = y(t),\quad \alpha\leqslant t \leqslant \beta\]
    因此有：
    \begin{align*}
        \int_{A}^{B}\vb{v}\cdot\dd{\vb{r}} &= \int_{L_{AB}}\grad\varphi\cdot\dd{\vb{r}} = \int_{L_{AB}}\varphi'_x\dd{x} + \varphi'_y\dd{y} \\
        &= \int_{\alpha}^{\beta}[\varphi'_x(x(t),y(t))x'(t) + \varphi'_y(x(t),y(t))y'(t)]\dd{t} \\
        &= \int_{\alpha}^{\beta}\dd{\varphi(x(t),y(t))} = \varphi(x(\beta),y(\beta)) - \varphi(x(\alpha),y(\alpha)) \\
        &= \varphi(B) - \varphi(A)
    \end{align*}
    由此得证.
\end{proof}


\section{向量场在曲面上的积分}
\subsection{双侧曲面及其定向}
\begin{definition}
    设有一张光滑曲面$S$，其在每一点$M$都有非零的法向量$\vb{n}(M)$和$-\vb{n}(M)$.对任意一点$M_0\in S$，取定其中一个法向量，不妨取$\vb{n}(M_0)$，作任意一条过$M_0$的闭曲线$L$，使$M$从$M_0$出发，在每一点取连续变化的法向量$\vb{n}(M)$.
    当$M$回到$M_0$时，若取到的法向量仍为$\vb{n}(M_0)$，则称$M_0$是曲面$S$的\overtext{双侧点}{two-sided point}.如果曲面上有一点是双侧的，则整个曲面称为\overtext{双侧曲面}{two-sided surface}或\overtext{可定向曲面}{orientable surface}.
\end{definition}

\begin{definition}
    在可定向曲面$S$上指定一侧为正面，或者说，在曲面上指定一个连续的法向量场$\vb{n}$为正向，则称曲面为\textbf{定向曲面}.
\end{definition}

对于定向曲面$S$，设单位法向量场$\vb{n}$为正向，若$S$能够以参数方程表示：
\[S:\vb{r}(u,v) = x(u,v)\vb{i} + y(u,v)\vb{j} + z(u,v)\vb{k}\]
如果$\vb{r}'_u\times\vb{r}'_v$与定向$\vb{n}$的指向相同，则称$(u,v)$是$S$的\textbf{正向参数}，相反则为反向参数.若$(u,v)$是反向参数，则$(v,u)$是正向参数.

有如下惯例：
\begin{itemize}
    \item 当曲面以$z=f(x,y)$显表示时，取与$z$轴正方向为锐角的方向为正向.所指向的一侧称为\textbf{上侧}，另一侧称为\textbf{下侧}.
    \item 对于封闭曲面，取向外的方向为正向.靠外的一侧称为\textbf{外侧}，另一面则称为\textbf{内侧}.
    \item 有时需要将有向曲面与有向曲线相协调，原则是曲面的取向与有向边界曲线的方向构成右手系.
    \item 若曲面由两个双侧曲面拼接而成，则也可以规定逆时针有向边界，并且用右手法则协调两个曲面的正向.
\end{itemize}

对于有向曲面，可以定义有向面积的概念：
\[\dd{\vb{S}}\coloneq \vb{n}\dd{S} = \frac{\vb{r}'_u\times\vb{r}'_v}{|\vb{r}'_u\times\vb{r}'_v|}|\vb{r}'_u\times\vb{r}'_v|\dd{u}\dd{v} = (\vb{r}'_u\times\vb{r}'_v)\dd{u}\dd{v}\]


\subsection{向量场在曲面上积分的计算}
\begin{definition}
    设$\vb{v}(M)$是定义在定向曲面$S$上的一个向量场，$S$的正向为单位法向量$\vb{n}$，则下列积分：
    \[\iint_S \vb{v}\cdot\dd{\vb{S}} \coloneq \iint_S\vb{v}\cdot\vb{n}\dd{S}\]
    称为向量场$\vb{v}$在有向曲面$S$上的\overtext{第二型曲面积分}{surface integral of the second kind}.

    特别地，若$S$是封闭曲面，积分也称为向量场$\vb{v}$通过封闭曲面$S$的\overtext{通量}{flux}.记作：
    \[\oiint_S \vb{v}\cdot\dd{\vb{S}}\]
\end{definition}

向量场在曲面上的积分具有以下性质：
\begin{enumerate}[property]
    \item \textbf{线性性}：若$\vb{v} = c_1\vb{v}_1 + c_2\vb{v}_2$，则有：
    \[\iint_S\vb{v}\cdot\dd{\vb{S}} = c_1\iint_S\vb{v}_1\cdot\dd{\vb{S}} + c_2\iint_S\vb{v}_2\cdot\dd{\vb{S}}\]
    \item \textbf{积分曲面的可加性}：若定向曲面$S$是由$S_1$和$S_2$协调地拼接而成，则有：
    \[\iint_S\vb{v}\cdot\dd{\vb{S}} = \iint_{S_1}\vb{v}\cdot\dd{\vb{S}} + \iint_{S_2}\vb{v}\cdot\dd{\vb{S}}\]
    \item \textbf{积分的方向性}：以$S^+$和$S^-$表示曲面的两个不同定向：
    \[\iint_{S^-}\vb{v}\cdot\vb{n^-}\dd{S} = -\iint_{S^+}\vb{v}\cdot\vb{n^+}\dd{S}\]
\end{enumerate}

\begin{theorem}
    设具有正向参数表示的定向光滑曲面$S$上有一光滑向量场$\vb{v}$，其在$S$上可积：
    \begin{align*}
        \iint_S\vb{v}\cdot\dd{\vb{S}} &= \iint_D\vb{v}\cdot\vb{r}'_u\times\vb{r}'_v\dd{u}\dd{v} = \iint_D\begin{vmatrix}
            P & Q & R \\ x'_u & y'_u & z'_u \\ x'_v & y'_v & z'_v
        \end{vmatrix}\dd{u}\dd{v} \\
        &= \iint_D\left[P\pdv{(y,z)}{(u,v)} + Q\pdv{(z,x)}{(u,v)} + R\pdv{(x,y)}{(u,v)}\right]\dd{u}\dd{v}
    \end{align*}
\end{theorem}

上式中，积分的方向隐含在三个Jacobi行列式中，有向面积元可以表示为：
\[\dd{\vb{S}} = (\vb{r}'_u\times\vb{r}'_v)\dd{u}\dd{v} = \left[\pdv{(y,z)}{(u,v)}\dd{u}\dd{v}\right]\vb{i} + \left[\pdv{(z,x)}{(u,v)}\dd{u}\dd{v}\right]\vb{j} + \left[\pdv{(x,y)}{(u,v)}\dd{u}\dd{v}\right]\vb{k}\]
以微分形式的外积简化上式，记：$\displaystyle\dd{x}\wedge\dd{y} = \pdv{(x,y)}{(u,v)}\dd{u}\dd{v}$，则有：
\[\dd{\vb{S}} = (\dd{y}\wedge\dd{z})\vb{i} + (\dd{z}\wedge\dd{x})\vb{j} + (\dd{x}\wedge\dd{y})\vb{k}\]
注意到，对于上述用外积所表达的式子，满足此前定义的微分形式外积的性质：
\[\dd{x}\wedge\dd{y} = \pdv{(x,y)}{(u,v)}\dd{u}\dd{v} = -\pdv{(y,x)}{(u,v)}\dd{u}\dd{v} = -\dd{y}\wedge\dd{x}\]
因而这也为二阶微分形式的三个生成元作出了一个几何解释：有向面积元分别在三个坐标平面上的投影.


在绝大多数教材中，都将$\dd{x}\wedge\dd{y}$简记为$\dd{x}\dd{y}$，因而第二型曲面积分也可以表示为：
\begin{align*}
    \iint_S\vb{v}\cdot\dd{\vb{S}} &= \iint_S P\dd{y}\wedge\dd{z} + Q\dd{z}\wedge\dd{x} + R\dd{x}\wedge\dd{y}\\
    &= \iint_S P\dd{y}\dd{z} + Q\dd{z}\dd{x} + R\dd{x}\dd{y}
\end{align*}
应当注意的是，上式中的$\dd{x},\dd{y},\dd{z}$不再能随意交换顺序.

特别地，若$S$是显式表示，不妨设曲面上侧为正向，$(x,y)$则为曲面的正向参数，于是有：
\[\iint_S \vb{v}\cdot\dd{\vb{S}} = \iint_D \begin{vmatrix}
    P & Q & R \\ 1 & 0 & f'_x \\ 0 & 1 & f'_y
\end{vmatrix}\dd{x}\dd{y} = \iint_D(-Pf'_x - Qf'_y + R)\dd{x}\dd{y}\]


\begin{question}
    计算曲面积分：
    \[\iint_S x^3\dd{y}\dd{z} + y^3\dd{z}\dd{x}\]
    其中，$S$是上半椭球面$\displaystyle\frac{x^2}{a^2}+\frac{y^2}{b^2}+\frac{z^2}{c^2} = 1(z\geqslant0)$，定向指向上侧.
\end{question}

\begin{solution}
    将上半椭球面$S$表示成参数方程：
    \[x = a\sin\theta\cos\varphi,\quad y = b\sin\theta,\quad z = c\cos\theta\]
    其中，$\theta,\varphi$的变化范围是一个矩形区域：
    \[D:0\leqslant\varphi\leqslant2\pi,\quad 0\leqslant\theta\leqslant\frac{\pi}{2}\]
    在区域$D$中，仅有关于$(x,y)$的Jacobi行列式恒大于$0$：
    \[\pdv{(x,y)}{(\theta,\varphi)} = ab\cos\theta\sin\theta > 0\]
    因而向量$\vb{r}'_\theta\times\vb{r}'_\varphi$指向$S$的正向，从而有：
    \begin{align*}
        \iint_S x^3\dd{y}\dd{z} &= \iint_S a^3\sin^3\theta\cdot\pdv{(y,z)}{(\theta,\varphi)}\dd{\theta}\dd{\varphi}\\
        &= a^3bc\int_{0}^{2\pi}\cos^4\varphi\dd{\varphi}\int_{0}^{\frac{\pi}{2}}\sin^5\theta\dd{\theta} = \frac{2}{5}\pi a^3bc
    \end{align*}
    同样有：
    \[\iint_S y^3\dd{z}\dd{x} = \frac{2}{5}\pi ab^3c\]
    由此求得：
    \[\iint_S x^3\dd{y}\dd{z} + y^3\dd{z}\dd{x} = \frac{2}{5}\pi abc(a^2+b^2)\]
\end{solution}


\section{Gauss定理和Stokes定理}
\subsection{Gauss定理}
考察光滑向量场：
\[\vb{v}(x,y,z) = P(x,y,z)\vb{i} + Q(x,y,z)\vb{j} + R(x,y,z)\vb{k}\]
通过空间中长方体$V = [0,a]\times[0,b]\times[0,c]$的表面$\partial V$外侧的通量.

设$S_1:x=a,y=y,z=z;S_2:x=0,y=y,z=z$，对应的法向量分别为$\vb{n}_1 = \vb{i},\vb{n}_2=-\vb{i}$，故而有：
\begin{gather*}
    \iint_{S_1}\vb{v}\cdot\dd{\vb{S}} = \int_{0}^{c}\int_{0}^{b}P(a,y,z)\dd{y}\dd{z} \\
    \iint_{S_2}\vb{v}\cdot\dd{\vb{S}} = -\int_{0}^{c}\int_{0}^{b}P(0,y,z)\dd{y}\dd{z}
\end{gather*}
因而在$S_1+S_2$上有：
\begin{align*}
    \iint_{S_1+S_2}\vb{v}\cdot\dd{\vb{S}} &= \int_{0}^{c}\int_{0}^{b}[P(a,y,z)-P(0,y,z)]\dd{y}\dd{z} \\
    &= \int_{0}^{c}\int_{0}^{b}\int_{0}^{a}\pdv{P(x,y,z)}{x}\dd{x}\dd{y}\dd{z} = \iiint_V\pdv{P(x,y,z)}{x}\dd{x}\dd{y}\dd{z}
\end{align*}
同理对于剩下四个面有：
\begin{gather*}
    \iint_{S_3+S_4}\vb{v}\cdot\dd{\vb{S}} = \iiint_V\pdv{Q(x,y,z)}{y}\dd{x}\dd{y}\dd{z} \\
    \iint_{S_5+S_6}\vb{v}\cdot\dd{\vb{S}} = \iiint_V\pdv{R(x,y,z)}{z}\dd{x}\dd{y}\dd{z}
\end{gather*}
将六个面合并，得到：
\[\oiint_S \vb{v}\cdot\dd{\vb{S}} = \iiint_V\left(\pdv{P}{x}+\pdv{Q}{y}+\pdv{R}{z}\right)\dd{x}\dd{y}\dd{z}\]
为了将Gauss定理推广到更广泛的区域中，可以模仿\cref{thm:GreenTheorem}\textbf{(Green定理)}的推广方法，先推广到一些更加普遍的子区域上.做如下证明准备：

定义以下三种类型的区域：
\begin{enumerate}[circ]
    \item \textbf{$Z$型区域}：$V = \{(x,y,z)\,|\,z\in[z_1(x,y),z_2(x,y)],(x,y)\in D\}$\\
    \textit{即一个中心轴平行于$z$轴，两端面由$z_1(x,y)$和$z_2(x,y)$显式表示的柱体.}
    \item \textbf{$Y$型区域}：$V = \{(x,y,z)\,|\,y\in[y_1(z,x),y_2(z,x)],(z,x)\in D\}$\\
    \textit{即一个中心轴平行于$y$轴，两端面由$y_1(z,x)$和$y_2(z,x)$显式表示的柱体.}
    \item \textbf{$Z$型区域}：$V = \{(x,y,z)\,|\,x\in[x_1(y,z),x_2(y,z)],(y,z)\in D\}$\\
    \textit{即一个中心轴平行于$x$轴，两端面由$x_1(y,z)$和$x_2(y,z)$显式表示的柱体.}
\end{enumerate}

\begin{theorem}
    设$R(x,y,z)\in C^1$，$\vb{v}_z = R(x,y,z)\vb{k}$，$V$是$Z$型区域，则：
    \[\iint_S \vb{v}_z\cdot\dd{\vb{S}} = \iiint_V\pdv{R}{z}\dd{x}\dd{y}\dd{z}\]
    其中$S = \partial V$，定向为$V$的外侧.
\end{theorem}

\begin{proof}
    显然，对于$Z$型区域，其侧面法向量与$z$轴垂直，故而其向量积分为$0$，因而只需考虑两个端面$S_1:z_1(x,y);S_2:z_2(x,y)$的积分：
    \begin{align*}
        \iint_S \vb{v}_z\cdot\dd{\vb{S}} &= \iint_{S_2}\vb{v}_z\cdot\dd{\vb{S}} + \iint_{S_1}\vb{v}_z\cdot\dd{\vb{S}} \\
        &= \iint_DR(x,y,z_2(x,y))\dd{x}\dd{y} - \iint_DR(x,y,z_1(x,y))\dd{x}\dd{y}\\
        &= \iint_D[R(x,y,z_2(x,y))-R(x,y,z_1(x,y))]\dd{x}\dd{y}\\
        &= \iint_D\left(\int_{z_1(x,y)}^{z_2(x,y)}\pdv{R}{z}\dd{z}\right)\dd{x}\dd{y}\\
        &= \iiint_V \pdv{R}{z}\dd{x}\dd{y}\dd{z}
    \end{align*}
\end{proof}
同理可以推出对$X$型、$Y$型区域类似的结论.

若一个区域$V$可以同时分解为上述类型的区域，并且相邻子区域公共边界上两侧的积分相互抵消，则有：
\begin{theorem}
    \textbf{Gauss定理}

    设$\vb{v} = P(x,y,z)\vb{i} + Q(x,y,z)\vb{j} + R(x,y,z)\vb{k}$是$V$上的光滑向量场，$V$是空间中分片光滑曲面围成的闭区域.如果$V$可以同时分解成有限个互不重叠的$X$型、$Y$型和$Z$型区域的并，则有：
    \[\oiint_S P\dd{y}\dd{z} + Q\dd{z}\dd{x} + R\dd{x}\dd{y} = \iiint_V\left(\pdv{P}{x}+\pdv{Q}{y}+\pdv{R}{z}\right)\dd{x}\dd{y}\dd{z}\]
    或者写为：
    \[\oiint_S \vb{v}\cdot\dd{\vb{S}} = \iiint_V\nabla\cdot\vb{v}\dd{V} = \iiint_V\operatorname{div}\vb{v}\dd{V}\]
\end{theorem}

Gauss定理为求取第二型曲面积分提供了一种新的方法：对于不封闭的曲面，可以将其用一个简单曲面封上，利用Gauss定理求解后再将其减去.

\begin{question}
    求积分：
    \[I = \iint_S x^2\dd{y}\dd{z} + y^2\dd{z}\dd{x} + z^2\dd{x}\dd{y}\]
    其中$S:x^2+y^2+z^2 = a^2,z\geqslant0$，定向为外侧.
\end{question}

\begin{solution}
    积分曲面是一个底面开口的半球面，为简化计算，用一个圆形底面$S_0:x^2+y^2\leqslant a^2,z=0$封住.记$\vb{v} = (x^2,y^2,z^2)$，由Gauss定理得：
    \begin{align*}
        I &= \iint_{S+S_0}\vb{v}\cdot\dd{\vb{S}} - \iint_{S_0}\vb{v}\cdot\dd{\vb{S}} = \oiint_{S+S_0}\vb{v}\cdot\dd{\vb{S}} - \iint_{S_0}\vb{v}\cdot\dd{\vb{S}}\\
        &= 2\iiint_V (x+y+z)\dd{x}\dd{y}\dd{z} - \iint_{S_0}z^2\dd{x}\dd{y}
    \end{align*}
    根据对称性，上式中关于$x$和$y$的三重积分等于$0$，而$S_0$上的面积分由于$z\equiv0$，故而也等于$0$，最终得到：
    \[I = 2\iiint_V z\dd{x}\dd{y}\dd{z} = \int_{0}^{a}z\dd{z}\underset{x^2+y^2\leqslant a^2-z^2}{\iint}\dd{x}\dd{y} = \frac{\pi a^4}{2}\]
\end{solution}


\subsection{Stokes定理}
\begin{theorem}
    \textbf{Stokes定理}

    设$\vb{v}=P(x,y,z)\vb{i} + Q(x,y,z)\vb{j} + R(x,y,z)\vb{k}$是$V$上的光滑向量场，$S$是分片具有二阶连续偏导数的光滑曲面，同时有界曲面$S$的边界记为曲线$L=\partial S$，则有：
    \[\!\!\underset{L}{\oint} P\dd{x}+Q\dd{y}+R\dd{z}\! = \!\!\underset{S}{\iint}\!\!\left(\pdv{R}{y}-\pdv{Q}{z}\right)\!\dd{y}\dd{z} + \!\left(\pdv{P}{z}-\pdv{R}{x}\right)\!\dd{z}\dd{x} + \!\left(\pdv{Q}{x}-\pdv{P}{y}\right)\!\dd{x}\dd{y}\!\!\!\!\]
    或者写为：
    \[\oint_L\vb{v}\cdot\dd{\vb{R}} = \iint_S\rot\vb{v}\cdot\dd{\vb{S}} = \iint_S \nabla\times\vb{v}\cdot\dd{\vb{S}}\]
\end{theorem}

\begin{proof}
    设$\mathbb{R}^3$中的$S$是定义域在$Ouv$平面上区域$D$的定向参数曲面：
    \[S:\vb{r}(u,v) = x(u,v)\vb{i} + y(u,v)\vb{j} + z(u,v)\vb{k},\quad (u,v)\in D \subset\mathbb{R}^2\]
    而区域$D$的边界曲线$\partial D$是定义域在$t$轴上$[\alpha,\beta]$区间的参数曲线：
    \[\partial D: u = u(t),\quad v = v(t),\quad t\in[\alpha,\beta]\subset\mathbb{R}\]
    故而有：
    \begin{gather*}
        L:\vb{r}(t) = \vb{r}(u(t),v(t)) = x(u(t),v(t))\vb{i} + y(u(t),v(t))\vb{j} + z(u(t),v(t))\vb{k},\quad t\in[\alpha,\beta]\\
        t\mapsto(u,v)\mapsto(x,y,z)\qquad [\alpha,\beta]\to \partial D \to L = \partial S
    \end{gather*}
    其中定向曲线$L$与定向曲面$S$相协调，由此有：
    \begin{gather*}
        \dd{x} = x'_u\dd{u} + x'_v\dd{v} = (x'_u u'_t + x'_v v'_t)\dd{t} \\
        \oint_L P\dd{x} = \int_{\alpha}^{\beta}P(x'_u u'_t + x'_v v'_t)\dd{t} = \int_{\partial D}(Px'_u)\dd{u} + (Px'_v)\dd{v}
    \end{gather*}
    由Green定理，将$\partial D$上的积分转到$D$上：
    \[\oint_L P\dd{x} = \iint_D\left[\pdv{(Px'_v)}{u} - \pdv{(Px'_u)}{v}\right]\dd{u}\dd{v}\]
    根据如下推导，$D$上的积分又可以转到$S$上：
    \begin{gather*}
        \pdv{(Px'_v)}{u} = (P'_x x'_u + P'_y y'_u + P'_z z'_u)x'_v + P x''_{vu} \\
        \pdv{(Px'_u)}{v} = (P'_x x'_v + P'_y y'_v + P'_z z'_v)x'_u + P x''_{uv}
    \end{gather*}
    \begin{align*}
        \iint_D\left[\pdv{(Px'_v)}{u} - \pdv{(Px'_u)}{v}\right]\dd{u}\dd{v} &= \iint_D\left[P'_z\pdv{(z,x)}{(u,v)} - P'_y\pdv{(x,y)}{(u,v)}\right]\dd{u}\dd{v} \\
        \oint_L P\dd{x} &= \iint_S P'_z\dd{z}\dd{x} - P'_y\dd{x}\dd{y}
    \end{align*}
    最终形成$S \to D \xlongrightarrow{\text{Green}} \partial D \to t \to L = \partial S$，同理有：
    \begin{gather*}
        \oint_L Q\dd{y} = \iint_S Q'_x\dd{x}\dd{y} - Q'_z\dd{y}\dd{z} \\
        \oint_L R\dd{z} = \iint_S R'_y\dd{y}\dd{z} - R'_x\dd{z}\dd{x}
    \end{gather*}
    由此得证.
\end{proof}

\begin{question}
    设曲线$L$是椭圆抛物面$z = 3x^2 + 4y^2$与椭圆柱面$4x^2+y^2 = 4y$的交线，从$z$轴的正方向看，$L$的方向为顺时针，求向量场$\vb{v} = y(z+1)\vb{i} + zx\vb{j} + (xy-z)\vb{k}$沿$L$的环量.
\end{question}

\begin{solution}
    设$D$为$Oxy$上的平面区域$4x^2+y^2\leqslant 4y$，$S$是曲面$z = 3x^2+4y^2,(x,y)\in D$的下侧，根据Stokes定理：
    \[\oint_L \vb{v}\cdot\dd{\vb{r}} = \iint_S\begin{vmatrix}
        \dd{y}\dd{z} & \dd{z}\dd{x} & \dd{x}\dd{y} \\
        \displaystyle\pdv{x} & \displaystyle\pdv{y} & \displaystyle\pdv{z} \\
        y(z+1) & zx & xy-z
    \end{vmatrix} = -\iint_S\dd{x}\dd{y} = \iint_D\dd{x}\dd{y} = 2\pi\]
\end{solution}
