\chapter{单变量函数的连续性}
\section{连续函数的基本概念}
\subsection{连续的定义}
\begin{definition}
    设$f(x)$在$x_0$的邻域内有定义且不为无穷，如果：
    \[\lim_{x\to x_0}f(x) = f(x_0)\]
    那么称$f(x)$在$x_0$处是\overtext{连续的}{continuous}.

    若$f(x)$在区间$I$上有定义并在$I$上任意一点都连续，则称$f(x)$在$I$上连续.
\end{definition}

用$\varepsilon-\delta$语言和集合来说，就是$\forall \varepsilon>0$，都$\exists \delta>0$使得：
\[f\left((x-\delta, x+\delta)\right) \subset \left(f(x_0)-\varepsilon, f(x_0)+\varepsilon\right)\]
这种表述在拓扑学中较为常见.

根据连续的定义，函数在一点的连续性与在这一点和这一点附近的值都有关系，这表明函数在一点连续是一个局部的性质.

\subsection{单侧连续}
\begin{definition}
    \begin{itemize}
        \item 若有$f(x_0-0)=f(x_0)$则称$f(x)$\textbf{左连续}.
        \item 若有$f(x_0+0)=f(x_0)$则称$f(x)$\textbf{右连续}.
    \end{itemize}
\end{definition}

\begin{theorem}
    函数在一点左连续并且右连续，当且仅当函数在这一点连续.
\end{theorem}


\subsection{间断的定义}
\begin{definition}
    设$f(x)$在$x_0$不连续，即\overtext{间断}{discontinuity}.

    \begin{itemize}
        \item 将左右极限存在且不为无穷的间断点称为\overtext{第一类间断点}{simple discontinuity}.
        \begin{itemize}
            \item 称满足$f(x_0-0)=f(x_0+0)\neq f(x_0)$或$f$在$x_0$处无定义的间断点称为\overtext{可去间断点}{removable discontinuity}.
            \item 称满足$f(x_0-0)\neq f(x_0+0)$的间断点为\overtext{跳跃间断点}{jump discontinuity}.
        \end{itemize}
        \item 将左右极限至少有一个不存在或者是无限的间断点称为\overtext{第二类间断点}{essential discontinuity}.\\
        具体包括\overtext{振荡间断点}{oscillating discontinuity}、\overtext{无穷间断点}{infinite discontinuity}.
    \end{itemize}
\end{definition}

\begin{extension}{处处不连续的函数}
    \textbf{Dirichlet函数}在处处都有定义，但是处处都不连续：
    \[D(x)=\begin{cases}
        1,\quad x \,\text{为有理数}\\
        0,\quad x \,\text{为无理数}
    \end{cases}\]
\end{extension}


\subsection{连续函数的性质}
\begin{enumerate}[property]
    \item \textbf{局部有界性：}若$f(x)$在某一点处连续，则$f(x)$在这一点的邻域上有界.
    \item 若$f(x),g(x)$在$x_0$处连续，则$\begin{cases}
        f(x)\pm g(x)\\
        f(x)g(x)\\
        \dfrac{f(x)}{g(x)}\,(g(x_0)\neq 0)
    \end{cases}$也在$x_0$处连续.
    \item 设$u=g(x)$在$x_0$处连续，$x=f(t)$在$t_0$处连续且$f(t_0)=x_0$，那么$g(f(t))$在$t_0$处连续.
    \item 若$y=f(x)$在$[a,b]$上连续，则$\eval{f}_{[a,b]}$存在反函数，当且仅当$f(x)$在$[a,b]$上严格单调.
    
    除此以外，$f(x)$的反函数$y=f^{-1}(x)$也是连续且严格单调的.
\end{enumerate}


\subsection{初等函数的连续性}
\begin{theorem}
    \begin{itemize}
        \item 基本初等函数在其定义域内连续.
        \item 经过基本初等函数由有限次四则运算和有限次复合运算所构成，能用一个解析式所描述的初等函数在其定义域内连续.
    \end{itemize}
\end{theorem}

\begin{question}
    证明$\lim\limits_{x\to 0}\dfrac{\ln(1+x)}{x}=1$.
\end{question}

\begin{proof}
    因为$\lim\limits_{x\to 0}(1+x)^{1/x}=\ee$，设：
    \[f(x)=\begin{cases}
        (1+x)^{1/x},\quad x \neq 0\\
        \ee,\quad x=0
    \end{cases}\]
    则$f(x)$在$x=0$处连续，又由于$\ln y$在$y=\ee$处连续，二者的复合函数$\ln(f(x))$在$x=0$处连续，进而有：
    \[\lim_{x\to 0}\frac{\ln(1+x)}{x} = \ln \lim_{x\to 0}f(x) = \ln \ee = 1\]
    得证.
\end{proof}

\begin{question}
    设$f(x)$在$|x|<\infty$连续且满足$f(x+y)=f(x)+f(y),\,x,y\in\mathbb{R}$.求$f(x)$的表达式.
\end{question}

\begin{solution}
    当$x=y$时，$f(2x)=2f(x)$，由此推出：
    \[f(nx)=nf(x) \quad\Rightarrow\quad f(\frac{x}{n})=\frac{1}{n}f(x) \quad\Rightarrow\quad f\left(\frac{m}{n}x\right)=\frac{m}{n}f(x)\]
    即对$\forall r \in \mathbb{Q}$都有$f(rx)=rf(x)$.

    令$y\equiv -x$，则有$f(0)=f(x)+f(-x)$；再令$x=y=0$，则有$f(0)=2f(0)$.

    由此推出$f(0)=0$，进而有$f(x)=-f(x)$.

    于是，$\forall x$，取有理数列$r_n \to x\,(n\to\infty)$，有：
    \[f(r_n)=r_n f(1)\]
    由于$f(x)$连续，故而有$f(r_n) = f(x) = f(1)\cdot x$，取$a=f(1)=\text{const}$，故而：
    \[f(x)=ax\,(a \in \mathbb{R})\]
\end{solution}


\section{闭区间上连续函数的性质}
\subsection{零点定理与介值定理}
首先需要明确一个概念，$f(x)$在闭区间$[a,b]$上连续指的是对$\forall x \in (a,b)$，$f(x)$在$x$处连续，对于$x=a$和$x=b$,$f(x)$分别左连续和右连续.记在$[a,b]$上所有连续函数的集合为$C[a,b]$.
\begin{theorem}
    \textbf{零点定理}

    设$f(x) \in C[a,b]$，有$f(a)f(b)<0$，则一定$\exists \xi \in (a,b)$使得$f(\xi)=0$.
\end{theorem}

\begin{proof}
    不妨设$f(a)<0<f(b)$，将$[a,b]$等分，如果$f\left(\dfrac{a+b}{2}\right)=0$，即得证.
    
    如果$f\left(\dfrac{a+b}{2}\right)\neq 0$，则区间$\left[a,\dfrac{a+b}{2}\right]$和$\dfrac{a+b}{2},b$必有一个其两端点异号，记异号的区间为$[a_1, b_1] \subset [a,b]$，有$f(a_1)<0<f(b_1)$，且$b_1-a_1=\dfrac{1}{2}(b-a)$.

    重复上述步骤，必定$\exists [a_n,b_n]$使得$f(a_n)<0<f(b_n)$并且有：
    \begin{gather*}
        b_n-a_n=\left(\frac{1}{2^n}\right)(b-a)\\
        [a_n,b_n]\subset[a_{n-1},b_{n-1}]\subset\cdots\subset[a,b]
    \end{gather*}
    根据区间套定理，必然有$a_n\to\xi,b_n\to\xi$，进一步有：
    \begin{gather*}
        f(a)<0<f(b)\\
        f(\xi)\leqslant 0 \leqslant f(\xi)
    \end{gather*}
    根据三明治定理和连续函数的性质，$f(\xi)=0$.
\end{proof}

\begin{theorem}
    \textbf{介值定理}

    设$f \in C[a,b]$，若有$f(a)\neq f(b)$则对任意$r$介于$f(a)$和$f(b)$之间，一定$\exists \xi \in (a,b)$使得$f(\xi)=r$.
\end{theorem}

\begin{proof}
    设$g(x)=f(x)-r$，显然有$g(x) \in C[a,b]$，对介于$f(a)$和$f(b)$之间的$r$，必然有；
    \[g(a)g(b) = (f(a)-r)(f(b)-r)<0\]
    根据零点定理，必然$\exists \xi \in (a,b)$使得$g(\xi)=f(\xi)-r=0$，即$f(\xi)=r$，得证.
\end{proof}

\begin{question}
    证明任意奇数次多项式必有实根.
\end{question}

\begin{proof}
    设$p(x)=a_{2n+1}x^{2n+1}+\cdots+a_0\,(a_{2n+1}\neq 0)$.

    当$x\to\infty$时，$p(x)\sim a_{2n+1}x^{2n+1}$，进而有$\lim\limits_{x\to\pm\infty}=\pm\infty$.

    显而易见的，左侧远处和右侧远处函数值异号，根据零点定理必有一点使得$p(x)=0$，即有实根.
\end{proof}

\begin{question}
    证明对于$f \in C[a,b]$，如果$f([a,b])\subset[a,b]$，则一定$\exists x_0 \in [a,b]$使得$f(x_0)=x_0$(称$x_0$为不动点).
\end{question}

\begin{proof}
    设$g(x)=f(x)-x$，必然有$\begin{cases}
        g(a)=f(a)-a\geqslant 0\\
        g(b)=f(b)-b\leqslant 0
    \end{cases}$，若两个等号至少有一个成立则结论显而易见，若两个等号都不成立则有$g(a)g(b)<0$，根据零点定理必有$x_0 \in [a,b]$使得$g(x_0)=f(x_0)-x_0=0$，得证.
\end{proof}


\subsection{有界性与最大最小值定理}
\begin{theorem}
    设$f(x)\in C[a,b]$，则$\eval{f}_{[a,b]}$有界.
\end{theorem}

\begin{proof}
    若$\eval{f}_{[a,b]}$无界，那么$\forall n>0$，一定$\exists x_n \in [a,b]$使得$|f(x_n)\geqslant n$.

    根据列紧性定理，有界数列必有收敛子列，即$\exists x_{n_k} \to x_n \in [a,b]$，对于子列，总有$n_k \geqslant k$，对于无穷的$k$，总是有：
    \[|f(x_{n_k})|\geqslant n_k \to \infty \quad\Rightarrow\quad |f(x_{n_k})|\to\infty\]
    另一方面又因为$f(x)\in C[a,b]$，有：
    \[\lim_{k\to\infty}f(x_{n_k}) = f(x_0) \neq \infty\]
    矛盾，故而有界.
\end{proof}

要注意开区间和闭区间的区别，比如$f=\dfrac{1}{x}$有$f \in C(0,1]$，但是$\eval{f}_(0,1]$无界.只有闭区间上的连续函数满足有界性.

\begin{theorem}
    \textbf{最值定理}

    设$f \in C[a,b]$，则$f$在$[a,b]$上一定能取到最大和最小值.
\end{theorem}

\begin{proof}
    因为$f \in C[a,b]$，则$f$必定有界，进而必定有上下确界.设有：
    \[\sup_{[a,b]}f(x) \equiv M \qquad\qquad \inf_{[a,b]}f(x) \equiv m\]
    则$\forall n>0$，$M-\dfrac{1}{n}$不再是上界，进而必定$\exists x_n \in [a,b]$满足：
    \[M \geqslant f(x_n) \geqslant M-\frac{1}{n}\]
    对于有界数列$x_n$，根据列紧性定理，必有收敛子列$x_{n_k} \to x^* \in [a,b]$，进而满足：
    \[M \geqslant f(x_{n_k}) \geqslant M-\frac{1}{n_k}\]
    取$k\to\infty$，根据三明治定理有$f(x_{n_k})\to M$，又根据连续性恰好有：
    \[\lim_{k\to\infty}f(x_{n_k}) = f(x^*) = M\]
    由于$M$是上界，则$f(x^*)=M$就是$\eval{f(x)}_{[a,b]}$的最大值.

    同理可证有最小值.
\end{proof}

\begin{theorem}
    设$f \in C[a,b]$，则$f([a,b])$是一个闭区间$\left[\inf\limits_{[a,b]}f(x),\sup\limits_{[a,b]}f(x)\right]$.
\end{theorem}

这个定理由上面的最值定理易证.


\subsection{一致连续性}
\begin{definition}
    设$f(x) \in C[a,b]$，对于$\forall \varepsilon > 0$，若$\exists \delta > 0$使得$\forall x_0 \in [a,b]$都在$|x-x_0|<\delta$时满足$|f(x)-f(x_0)|<\varepsilon$，则称$f(x)$在$[a,b]$上\overtext{一致连续}{uniformly continuous}.
\end{definition}

即能否有一个在整个区间上\textbf{一致}的$\delta$满足连续性条件，而不是在各个点取不一样的$\delta$来满足连续性条件，即逐点连续.

\begin{question}
    求使得$f(x)=\sin x$在其定义域上一致连续的$\delta$.
\end{question}

\begin{solution}
    首先根据定义，如果满足一致连续，应当有$\forall x',x'' \in I$满足$|x'-x''|<\delta$，还满足$|\sin x' - \sin x''|<\varepsilon$，注意到：
    \[|\sin x' - \sin x''| = 2\left|\sin\frac{x'-x''}{2}\cos\frac{x'+x''}{2}\right|\leqslant|x'-x''|<\delta\]
    故而，令$\delta=\varepsilon$，即可满足一致连续条件.
\end{solution}

\begin{theorem}
    在闭区间上连续的函数一定在这个闭区间内一致连续.
\end{theorem}

\begin{proof}
    假设$\eval{f}_{[a,b]}$不一致连续.

    这意味着$\exists \varepsilon_0>0$，对$\forall \delta\equiv\frac{1}{n}>0$满足：$\forall x_n',x_n'' \in [a,b]$在$|x_n'-x_n''|<\delta=\frac{1}{n}$时有$|f(x_n')-f(x_n'')|\geqslant \varepsilon_0$，即没有一个适用于所有点的$\delta$.

    根据列紧性定理，必有收敛子列$x_{n_k}'\to x'\,(k\to\infty)$.由于：
    \[0<|x_{n_k}'-x_{n_k}''|<\frac{1}{n_k}\to 0\]
    根据三明治定理，必然有$x_{n_k}'' \to x'$.又根据$f$连续，有：
    \[\lim_{k\to\infty}\left(f(x_{n_k}')-f(x_{n_k}'')\right)=0 \geqslant \varepsilon_0\]
    这与$\varepsilon_0>0$是矛盾的，所以假设不成立.
\end{proof}

这并不是说不在闭区间上连续的函数就不能一致连续，假若在无穷远处函数具有不剧烈发散等良好的性质，依然可以保持一致连续.

\begin{question}
    证明：若$\eval{f}_{[a,+\infty)}$连续，且在$x\to+\infty$时$f(x)\to l$是有限值，则$f$在$[a,+\infty)$上一致连续.
\end{question}

\begin{proof}
    要证明一致连续，就表明对于一个待求取的区间宽度$\delta$，它在任何地方对于两个间距小于$\delta$的点$x',x''$都满足$|f(x')-f(x'')|<\varepsilon$.

    已知在远处$f$趋于收敛，根据Cauchy收敛准则，一定存在一个边界$M$使得$x',x''>M$后，对于小于等于任意间距的两个点，总有$|f(x')-f(x'')|<\varepsilon$，即在远处一致连续，不妨设这个步长为$1$.

    注意到，「远处」指的是区间$(M,+\infty)$.

    进一步考虑近处的一致连续性，假如把「近处」设定为某个闭区间，由于$f$是连续的，那么它在这个闭区间上一定是一致连续的.

    证明的主要目的是找到一个在任何地方都适用的$\delta$，问题在于，使得远处一致连续的$1$完全可能使得近处不一致连续，因为近处尚未趋于收敛，函数值的抖动较大.

    于是，取二者的最小值作为全局的$\delta$，即$\delta=\min\{\delta_1,1\}$，这使得$\delta$在「远处」和「近处」两个区间里都满足一致连续条件.

    现在，只需要把两个区间接上，就可以完成证明.

    倘若将「近处」取为$[a,M]$会导致一个问题，若是间距为$\delta$的两个点各自落在「远处」和「近处」两个区间里，那么两边关于一致连续的要求都不能管到另一个点，这使得一致连续的判定失效了.

    这时，只需要使得两个区间形成一个「共同管理」的交叠部分，这个部分的宽度设置为两点最大的可能间距$1$，即「近处」取为$[a,M+1]$，这就保证了关于一致连续性的判定标准能够发生平稳过渡.
\end{proof}
