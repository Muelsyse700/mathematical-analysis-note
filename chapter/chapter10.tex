\chapter{多变量函数的重积分}
\section{二重积分}
\subsection{平面区域的面积}
记平面点集$D$的面积为$\sigma(D) \geqslant 0$，并满足：
\begin{enumerate}[property]
    \item 约定面积的单位为边长为1的正方形面积，边长为$l$的正方形面积为$l^2$.\\
    \textit{这规定了面积数值的尺度，使得边长为$l$的正方形面积是$l^2$而不是$100l^2$.}
    \item 如果$D$由无公共内点的有限个集合组成，则其满足可加性$\sigma(D) = \sum\sigma(D_i)$.
    \textit{这确保了此后进行分割操作的合理性.}
\end{enumerate}
设$D$是一个有界集合，则必然存在一个正方形$[a,b]\times[a,b]$能够将其「框住」，即$D\subset[a,b]\times[a,b]$.对正方形做分割：
\[T:a = x_0 < x_1 < \cdots < x_n = b\]
由此得到$n^2$个小正方形.记完全包含于$D$内的小正方形面积之和为$\sigma_T^-(D)$，所有与$D$有公共点的小正方形面积为$\sigma_T^+(D)$.

随着分割的加细，$\sigma_T^-$单调递增且有上界$(b-a)^2$，$\sigma_T^+$单调递减且有下界$0$.根据\cref{thm:MonotoneBounded}\textbf{(单调有界原理)}，当$\left\|T\right\|\to0$时，$\sigma_T^-(D)$和$\sigma_T^+(D)$必有极限.

    \caption{二维区域的分割}
\end{figure}

\begin{definition}
    设$D$是有界的平面点集.令$D\subset[a,b]\times[a,b]$，如果对于任意分割$T:a = x_0 < x_1 < \cdots < x_n = b$有下列极限相等：
    \[\lim_{\left\|T\right\|\to0}\sigma_T^+(D) = \lim_{\left\|T\right\|\to0}\sigma_T^-(D)\]
    则称$D$是\textbf{Jordan可测的}.上述极限的值称为$D$的\textbf{测度}，当极限值为$0$时，称$D$为\textbf{零测集}.
\end{definition}

\begin{theorem}
    $D$是Jordan可测的，当且仅当其边界$\partial D$的测度为$0$.
\end{theorem}

\begin{proof}
    易证$\partial D$是零测集时，$D$可测：
    \[0 \leqslant \sigma_T^+(D) - \sigma_T^-(D) \leqslant \sigma_T^+(\partial D)\]
    反向的证明暂略.
\end{proof}


\subsection{二重积分的基本概念与性质}
\begin{definition}
    定义对区域$D$的任意分割$T$指的是将$D$分成$n$个互不重叠且可测的区域$D_i$，并记：
    \[\left\|T\right\| = \max_{1\leqslant i \leqslant n}\operatorname{diam}D_i\]
\end{definition}
应当指出，这里分割的区域并不一定是正方形的.由此构造二维的Riemann和，并有二重Riemann积分的定义：
\begin{definition}
    若存在实数$A$满足对$\forall \varepsilon > 0$，$\exists\delta>0$使得在$\left\|T\right\|<\delta$时，对$\forall(\xi_i,\eta_i)\in D_i$，有：
    \[\left|\sum_{i=1}^{n}f(\xi_i,\eta_i)\sigma(D_i) - A\right|<\varepsilon\]
    则称函数$f(x,y)$在$D$上可积，记为：
    \[A = \iint_D f(x,y)\dd{\sigma}\]
\end{definition}

\begin{theorem}
    $D$上可积函数必有界，反之不然.
\end{theorem}
证明同单变量情形下的\cref{thm:IntegrableImpliesBounded}.

同单变量情形一样，二重积分也可以定义相应的Darboux上下和及振幅之和：
\begin{gather*}
    M_i = \sup\{f(x,y)\,|\,(x,y)\in D_i\},\quad m_i = \inf\{f(x,y)\,|\,(x,y)\in D_i\},\quad 1\leqslant i \leqslant n \\
    \overline{S}(T) \coloneq \sum_{i=1}^{n}M_i\sigma(D_i) \geqslant m \sigma(D),\quad \underline{S}(T) \coloneq \sum_{i=1}^{n}m_i\sigma(D_i) \leqslant M\sigma(D) \\
    \omega(T) \coloneq \sum_{i=1}^{n}(M_i - m_i)\sigma(D_i) = \overline{S}(T) - \underline{S}(T)
\end{gather*}
并且总是有$\displaystyle\inf_T\{\overline{S}(T)\} \geqslant \sup_T\{\underline{S}(T)\}$.由此有下列结论：
\begin{theorem}
    设$f(x,y)$在$D$上有界.
    \begin{enumerate}[circ]
        \item $f(x,y)$在$D$上可积，当且仅当：
        \[\inf_T\{\overline{S}(T)\} = \sup_T\{\underline{S}(T)\} \quad\text{或}\quad \inf_T\{\omega(T)\} = 0\]
        \item \textbf{(连续必可积)}如果$f(x,y)$在$D$上连续，则其在$D$上可积.\\
        \textit{如果$f(x,y)$的不连续点分布在$D$中可测且测度为$0$的点集上，如有限个孤立点或有限条光滑曲线，那么$f(x,y)$在$D$上也可积.}
        \item 如果在$D$上使$f(x,y) \neq g(x,y)$的点分布在$D$中测度为$0$的点集上，那么$f$和$g$具有相同的可积性且在可积时有：
        \[\iint_D f = \iint_D g\]
        \textit{即在测度为$0$的点集上任意地改变函数值不改变函数的可积性，也不改变积分的结果.}
    \end{enumerate}
\end{theorem}

\begin{theorem}
    设$f(x,y)$和$g(x,y)$在$D$上可积.
    \begin{enumerate}[circ]
        \item 对任意的常数$c_1,c_2$有：
        \[\iint_D (c_1 f + c_2 g) = c_1 \iint_D f + c_2 \iint_D g\]
        \item 设$D$是两个可测点集$D_1,D_2$无公共内点的拼接，若$f$在$D_1,D_2$上分别可积，则其在$D = D_1 \cup D_2$上也可积，并且满足：
        \[\iint_{D_1 \cup D_2} f = \iint_{D_1} f + \iint_{D_2} f\]
        \item $\varphi(x,y) = f(x,y)\cdot g(x,y)$在$D$上仍可积.
        \item 若在$D$上$f(x,y) \geqslant g(x,y)$，则：
        \[\iint_D f \geqslant \iint_D g\]
        若$D$上有$m \leqslant f(x,y) \leqslant M$，则有：
        \[m\sigma(D) \leqslant \iint_D f \leqslant M\sigma(D)\]
        \item $f$在$D$上可积则$|f|$在$D$上也可积，并有：
        \[\left|\iint_D f\right| \leqslant \iint_D |f|\]
        \item \textbf{(积分中值定理)}若$D$是连通闭区域，$f$在$D$上连续，则必有一点$P \in D$使得：
        \[\iint_D f = f(P)\sigma(D)\]
    \end{enumerate}
\end{theorem}
