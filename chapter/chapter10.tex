\chapter{多变量函数的重积分}
\section{二重积分}
\subsection{平面区域的面积}
记平面点集$D$的面积为$\sigma(D) \geqslant 0$，并满足：
\begin{enumerate}[property]
    \item 约定面积的单位为边长为1的正方形面积，边长为$l$的正方形面积为$l^2$.\\
    \textit{这规定了面积数值的尺度，使得边长为$l$的正方形面积是$l^2$而不是$100l^2$.}
    \item 如果$D$由无公共内点的有限个集合组成，则其满足可加性$\sigma(D) = \sum\sigma(D_i)$.
    \textit{这确保了此后进行分割操作的合理性.}
\end{enumerate}
设$D$是一个有界集合，则必然存在一个正方形$[a,b]\times[a,b]$能够将其「框住」，即$D\subset[a,b]\times[a,b]$.对正方形做分割：
\[T:a = x_0 < x_1 < \cdots < x_n = b\]
由此得到$n^2$个小正方形.记完全包含于$D$内的小正方形面积之和为$\sigma_T^-(D)$，所有与$D$有公共点的小正方形面积为$\sigma_T^+(D)$.

随着分割的加细，$\sigma_T^-$单调递增且有上界$(b-a)^2$，$\sigma_T^+$单调递减且有下界$0$.根据\cref{thm:MonotoneBounded}\textbf{(单调有界原理)}，当$\left\|T\right\|\to0$时，$\sigma_T^-(D)$和$\sigma_T^+(D)$必有极限.

    \centering
    \includegraphics[width=\textwidth]{region-partition/fig.pdf}
    \caption{二维区域的分割}
\end{figure}

\begin{definition}
    设$D$是有界的平面点集.令$D\subset[a,b]\times[a,b]$，如果对于任意分割$T:a = x_0 < x_1 < \cdots < x_n = b$有下列极限相等：
    \[\lim_{\left\|T\right\|\to0}\sigma_T^+(D) = \lim_{\left\|T\right\|\to0}\sigma_T^-(D)\]
    则称$D$是\textbf{Jordan可测的}.上述极限的值称为$D$的\textbf{测度}，当极限值为$0$时，称$D$为\textbf{零测集}.
\end{definition}

\begin{theorem}
    $D$是Jordan可测的，当且仅当其边界$\partial D$的测度为$0$.
\end{theorem}

\begin{proof}
    易证$\partial D$是零测集时，$D$可测：
    \[0 \leqslant \sigma_T^+(D) - \sigma_T^-(D) \leqslant \sigma_T^+(\partial D)\]
    反向的证明暂略.
\end{proof}


\subsection{二重积分的基本概念与性质}
\begin{definition}
    定义对区域$D$的任意分割$T$指的是将$D$分成$n$个互不重叠且可测的区域$D_i$，并记：
    \[\left\|T\right\| = \max_{1\leqslant i \leqslant n}\operatorname{diam}D_i\]
\end{definition}
应当指出，这里分割的区域并不一定是正方形的.由此构造二维的Riemann和，并有二重Riemann积分的定义：
\begin{definition}
    若存在实数$A$满足对$\forall \varepsilon > 0$，$\exists\delta>0$使得在$\left\|T\right\|<\delta$时，对$\forall(\xi_i,\eta_i)\in D_i$，有：
    \[\left|\sum_{i=1}^{n}f(\xi_i,\eta_i)\sigma(D_i) - A\right|<\varepsilon\]
    则称函数$f(x,y)$在$D$上可积，记为：
    \[A = \iint_D f(x,y)\dd{\sigma}\]
\end{definition}

\begin{theorem}
    $D$上可积函数必有界，反之不然.
\end{theorem}
证明同单变量情形下的\cref{thm:IntegrableImpliesBounded}.

同单变量情形一样，二重积分也可以定义相应的Darboux上下和及振幅之和：
\begin{gather*}
    M_i = \sup\{f(x,y)\,|\,(x,y)\in D_i\},\quad m_i = \inf\{f(x,y)\,|\,(x,y)\in D_i\},\quad 1\leqslant i \leqslant n \\
    \overline{S}(T) \coloneq \sum_{i=1}^{n}M_i\sigma(D_i) \geqslant m \sigma(D),\quad \underline{S}(T) \coloneq \sum_{i=1}^{n}m_i\sigma(D_i) \leqslant M\sigma(D) \\
    \omega(T) \coloneq \sum_{i=1}^{n}(M_i - m_i)\sigma(D_i) = \overline{S}(T) - \underline{S}(T)
\end{gather*}
并且总是有$\displaystyle\inf_T\{\overline{S}(T)\} \geqslant \sup_T\{\underline{S}(T)\}$.由此有下列结论：
\begin{theorem}
    设$f(x,y)$在$D$上有界.
    \begin{enumerate}[circ]
        \item $f(x,y)$在$D$上可积，当且仅当：
        \[\inf_T\{\overline{S}(T)\} = \sup_T\{\underline{S}(T)\} \quad\text{或}\quad \inf_T\{\omega(T)\} = 0\]
        \item \textbf{(连续必可积)}如果$f(x,y)$在$D$上连续，则其在$D$上可积.\\
        \textit{如果$f(x,y)$的不连续点分布在$D$中可测且测度为$0$的点集上，如有限个孤立点或有限条光滑曲线，那么$f(x,y)$在$D$上也可积.}
        \item 如果在$D$上使$f(x,y) \neq g(x,y)$的点分布在$D$中测度为$0$的点集上，那么$f$和$g$具有相同的可积性且在可积时有：
        \[\iint_D f = \iint_D g\]
        \textit{即在测度为$0$的点集上任意地改变函数值不改变函数的可积性，也不改变积分的结果.}
    \end{enumerate}
\end{theorem}

\begin{theorem}
    设$f(x,y)$和$g(x,y)$在$D$上可积.
    \begin{enumerate}[circ]
        \item 对任意的常数$c_1,c_2$有：
        \[\iint_D (c_1 f + c_2 g) = c_1 \iint_D f + c_2 \iint_D g\]
        \item 设$D$是两个可测点集$D_1,D_2$无公共内点的拼接，若$f$在$D_1,D_2$上分别可积，则其在$D = D_1 \cup D_2$上也可积，并且满足：
        \[\iint_{D_1 \cup D_2} f = \iint_{D_1} f + \iint_{D_2} f\]
        \item $\varphi(x,y) = f(x,y)\cdot g(x,y)$在$D$上仍可积.
        \item 若在$D$上$f(x,y) \geqslant g(x,y)$，则：
        \[\iint_D f \geqslant \iint_D g\]
        若$D$上有$m \leqslant f(x,y) \leqslant M$，则有：
        \[m\sigma(D) \leqslant \iint_D f \leqslant M\sigma(D)\]
        \item $f$在$D$上可积则$|f|$在$D$上也可积，并有：
        \[\left|\iint_D f\right| \leqslant \iint_D |f|\]
        \item \textbf{(积分中值定理)}若$D$是连通闭区域，$f$在$D$上连续，则必有一点$P \in D$使得：
        \[\iint_D f = f(P)\sigma(D)\]
    \end{enumerate}
\end{theorem}


\subsection{二重积分的计算}
对于矩形区域$D = [a,b]\times[c,d]$，设有$f(x,y)$在$D$上可积，由两个分割得到一个矩形分割：
\begin{gather*}
    \begin{cases}
        T_x: a = x_0 < x_1 < \cdots < x_n = b \\
        T_y: c = y_0 < y_1 < \cdots < y_m = d
    \end{cases}\\
    \Rightarrow\quad T:D_{ij} = [x_{i-1},x_i]\times[y_{j-1},y_j],\quad i = 1,\cdots,n;\,j = 1,\cdots,m
\end{gather*}
特别有$\sigma(D_{ij})=(x_i-x_{i-1})(y_j-y_{j-1}) = \Delta x_i \Delta y_i$.

在$D_{ij}$中任取一点$M_{ij}(\xi_i,\eta_j)$，则函数$f(x,y)$在$D$中的Riemann和为：
\[S(T,M) = \sum_{i,j}f(M_{ij})\Delta x_i \Delta y_j\]
由于$f$在$D$上可积，于是$\forall\varepsilon > 0$，$\exists\delta > 0$，在$\left\|T\right\|<\delta$时，对$\forall M_{ij}\in D_{ij}$有：
\begin{gather*}
    A = \lim_{\left\|T\right\|\to0}\sum_{i,j}^{n,m}f(M_{ij})\Delta x_i \Delta y_j \quad\Leftrightarrow\quad A - \frac{\varepsilon}{2} < \sum_{i,j}^{n,m}f(M_{ij})\Delta x_i \Delta y_j < A + \frac{\varepsilon}{2} \\
    \Rightarrow\quad A-\frac{\varepsilon}{2} < \sum_{j=1}^{m}\underbrace{\left[\sum_{i=1}^{n}f(\xi_i,\eta_j)\Delta x_i\right]}_\text{Riemann和}\Delta y_j < A + \frac{\varepsilon}{2}
\end{gather*}
若对于任何固定的$y \in [c,d]$，$f(x,y)$作为关于$x$的函数在$[a,b]$上可积，则有：
\[\lim_{\left\|T_x\right\|\to0}\sum_{i=1}^{n}f(\xi_i,\eta_j)\Delta x_i = \int_{a}^{b}f(x,\eta_j)\dd{x} = \varphi(\eta_j)\]
此外，注意到：
\begin{gather*}
    \left\|T\right\| = \max_{\substack{1\leqslant i \leqslant n \\ 1\leqslant j \leqslant m}}\operatorname{diam} D_{ij} = \sqrt{(\Delta x_i)^2 + (\Delta y_j)^2} < \Delta x_i + \Delta y_j\\
    \Rightarrow\quad \max\left\{\left\|T_x\right\|,\left\|T_y\right\|\right\} < \left\|T\right\| < \max_{1\leqslant i \leqslant n}\Delta x_i + \max_{1\leqslant j \leqslant m}\Delta y_j = \left\|T_x\right\| + \left\|T_y\right\|
\end{gather*}
因而在$\left\|T_x\right\|\to0$时，由于$\left\|T\right\| < \delta$，故而只要$\left\|T_y\right\|<\delta$就有：
\[A - \varepsilon < A - \frac{\varepsilon}{2} \leqslant \sum_{j=1}^{m}\varphi(\eta_j)\Delta y_j \leqslant A + \frac{\varepsilon}{2} < A + \varepsilon\]
上式表明$\varphi(y)$在$[c,d]$上可积，且有：
\[\int_{c}^{d}\varphi(y)\dd{y} = \int_{c}^{d}\left[\int_{a}^{b} f(x,y) \dd{x}\right]\dd{y} = \iint_D f(x,y)\dd{x}\dd{y}\]
由此得到以下定理：
\begin{theorem}
    设$f(x,y)$在$D = [a,b]\times[c,d]$上可积.
    \begin{enumerate}[circ]
        \item 若对$\forall y \in [c,d]$，$f(x,y)$作为$x$的函数在$[a,b]$上可积，则函数$\varphi(y) = \int_{a}^{b}f(x,y)\dd{x}$关于$y$在$[c,d]$上可积，并且有：
        \[\int_{c}^{d}\varphi(y)\dd{y} = \int_{c}^{d}\dd{y}\int_{a}^{b}f(x,y)\dd{x} = \iint_D f(x,y)\dd{x}\dd{y}\]
        \item 若对$\forall x \in [a,b]$，$f(x,y)$作为$y$的函数在$[c,d]$上可积，则函数$\varphi(x) = \int_{c}^{d}f(x,y)\dd{y}$关于$x$在$[a,b]$上可积，并且有：
        \[\int_{a}^{b}\varphi(x)\dd{x} = \int_{a}^{b}\dd{x}\int_{c}^{d}f(x,y)\dd{y} = \iint_D f(x,y)\dd{x}\dd{y}\]
        \item 若$f(x,y)$同时满足\circnum{1}和\circnum{2}，则积分的次序可交换：
        \[\iint_D f(x,y)\dd{x}\dd{y} = \int_{c}^{d}\dd{y}\int_{a}^{b}f(x,y)\dd{x} = \int_{a}^{b}\dd{x}\int_{c}^{d}f(x,y)\dd{y}\]
    \end{enumerate}
\end{theorem}
上述将二重积分转化为先对一个变量做单变量积分，再对另一个变量做单变量积分的过程称为\textbf{累次积分}.

前述讨论都是基于矩形积分区域进行的讨论.接下来讨论其它形状的区域：
\begin{theorem}
    设$f(x,y)$在有界闭区域$D$上可积.
    \begin{enumerate}[circ]
        \item 设$D = \{(x,y)\,|\,a \leqslant x \leqslant b,y_1(x)\leqslant y \leqslant y_2(x)\}$，其中$y_1(x),y_2(x)$在$[a,b]$上连续.若对$\forall x \in[a,b]$，$f(x,y)$在$[y_1(x),y_2(x)]$上关于$y$可积，则有：
        \[\iint_D f(x,y)\dd{x}\dd{y} = \int_{a}^{b}\dd{x}\int_{y_1(x)}^{y_2(x)}f(x,y)\dd{y}\]
        \item 设$D = \{(x,y)\,|\,x_1(y) \leqslant x \leqslant x_2(y),c \leqslant y \leqslant d\}$，其中$x_1(y),x_2(y)$在$[c,d]$上连续.若对$\forall y \in[c,d]$，$f(x,y)$在$[x_1(y),x_2(y)]$上关于$x$可积，则有：
        \[\iint_D f(x,y)\dd{x}\dd{y} = \int_{c}^{d}\dd{y}\int_{x_1(y)}^{x_2(y)}f(x,y)\dd{x}\]
        \item 若$f(x,y)$连续且积分区域$D$可以分别表述为\circnum{1}和\circnum{2}中的两种形式，则累次积分顺序可交换：
        \[\iint_D f(x,y)\dd{x}\dd{y} = \int_{c}^{d}\dd{y}\int_{x_1(y)}^{x_2(y)}f(x,y)\dd{x} = \int_{a}^{b}\dd{x}\int_{y_1(x)}^{y_2(x)}f(x,y)\dd{y}\]
    \end{enumerate}
\end{theorem}

\begin{proof}
    \circnum{1}的证明：设$D \subset [a,b]\times[c,d]$，对于$D$上的函数$f(x,y)$，设有如下函数$f^*$在$[a,b]\times[c,d]$上：
    \[f^*(x,y) = \begin{cases}
        f(x,y),\quad (x,y)\in D \\
        0,\quad (x,y)\notin D
    \end{cases}\]
    记$D^* = [a,b]\times[c,d] \setminus D$，有：
    \[\iint_{[a,b]\times[c,d]}f^*(x,y)\dd{x}\dd{y} = \iint_D\!\! f^*(x,y)\dd{x}\dd{y} + \iint_{D^*}\!\!f^*(x,y)\dd{x}\dd{y} = \iint_D\!\! f(x,y)\dd{x}\dd{y}\]
    将积分区域变换为矩形后，模仿矩形区域积分的有关证明即可得证.

    关于\circnum{2}和\circnum{3}的证明同理，且如果积分区域可以通过分解变为若干块满足\circnum{1}或\circnum{2}要求的子区域，各子区域分块积分之和即为整个区域的积分.
\end{proof}

\begin{question}
    计算：$\displaystyle\int_{1}^{2}y\dd{y}\int_{y}^{2}\frac{\sin x}{x^2-1}\dd{x}$.
\end{question}

\begin{solution}
    由于原式的积分顺序需要先求$\dfrac{\sin x}{x^2-1}$的原函数，十分复杂，考虑先将积分区域表示出来，改换积分顺序.
    \begin{align*}
        \text{原式} &= \iint_{\,\substack{1\leqslant y \leqslant 2 \\ y\leqslant x \leqslant 2}}\frac{y \sin x}{x^2-1}\dd{x}\dd{y} = \int_{1}^{2}\frac{\sin x}{x^2-1}\dd{x}\int_{1}^{x}y\dd{y} \\
        &= \frac{1}{2}\int_{1}^{2}\frac{\sin x}{x^2-1}(x^2-1)\dd{x} = \frac{1}{2}(\cos 1 - \cos 2)
    \end{align*}
\end{solution}

\begin{question}
    求由$x^2 + y^2 \leqslant a^2$和$x^2 + z^2 \leqslant a^2$相交部分的立体的体积$V$.
\end{question}

\begin{solution}
    相交所得到的图形在在第一卦限表现为底部为四分之一圆$D:x^2 + y^2 \leqslant a^2,x\in[0,a], y\in[0,a]$，顶部是圆柱侧面的一部分$z = \sqrt{a^2 - x^2},x\in[0,a]$.
    \begin{align*}
        \frac{1}{8}V &= \iint_D \sqrt{a^2 - x^2}\dd{x}\dd{y} = \int_{0}^{a}\sqrt{a^2-x^2}\dd{x}\int_{0}^{\sqrt{a^2-x^2}}\dd{y} \\
        &= \int_{0}^{a}(a^2-x^2)\dd{x} = \frac{2}{3}a^3
    \end{align*}
\end{solution}


\section{二重积分的换元}
二重积分的换元法，实际上是在一个新的被积区域上作矩形分割，将其映射到原积分区域的非矩形分割上.
\subsection{坐标曲线和面积元素}
设$D$上的$(x,y)$可换元为$(u,v)$，即设有：
\[(u,v)\mapsto(x,y):x = x(u,v),\quad y=y(u,v),\quad (u,v)\in D'\]
通过固定$u$或$v$，可以给出$D'$中的矩形分割，对应$D$中由曲线族分割的网格.则$D'$中的一个小矩形$[u,u+\Delta u]\times[v,v+\Delta v]$对应$D$中的一个小曲边四边形：
\begin{align*}
    (u,v)&\mapsto M_1(x_1,y_1) &(u+\Delta u,v)\mapsto M_2(x_2,y_2)\\
    (u,v+\Delta v)&\mapsto M_3(x_3,y_3) &(u+\Delta u,v+\Delta v)\mapsto M_4(x_4,y_4)
\end{align*}

\begin{figure}[H]
    \centering
    \includegraphics[scale=0.28]{double-integral-substitution/fig.pdf}
    \caption{二重积分换元的映射关系}
\end{figure}

当分割变细时，曲边四边形的面积可以表示为：
\[\Delta\sigma \approx \left|\va{M_1M_2}\times\va{M_1M_4}\right| = |(x_2-x_1)(y_4-y_1) - (x_4-x_1)(y_2-y_1)|\]
假设$x(u,v),y(u,v)$可微，则有：
\begin{align*}
    x_2-x_1 &= \pdv{x}{u}\Delta u + o(\Delta u) &x_4-x_1 = \pdv{x}{v}\Delta v + o(\Delta v) \\
    y_2-y_1 &= \pdv{y}{u}\Delta u + o(\Delta u) &y_4-y_1 = \pdv{y}{v}\Delta v + o(\Delta v)
\end{align*}
略去高阶无穷小量，得到小曲边四边形面积$\Delta\sigma$与小矩形面积$\Delta u \Delta v$之间的关系：
\[\Delta\sigma \approx\left|\pdv{x}{u}\pdv{y}{v} - \pdv{y}{u}\pdv{x}{v}\right|\Delta u \Delta v = \left|\pdv{(x,y)}{(u,v)}\right|\Delta u \Delta v\]
上式中的Jacobi矩阵描述了坐标变换导致的面积变化率，显然应当有$\displaystyle\pdv{(x,y)}{(u,v)}\neq 0$.


\subsection{二重积分的换元}
设$f(x,y)\in C(D)$，对区域$D'$作矩形分割.由于有小区域$D'_{ij}$到$D_{ij}$的映射：
\[\sigma(D_{ij}) \approx \left|\pdv{(x,y)}{(u,v)}\right|_{(u_i,v_j)} \Delta u_i \Delta v_j\]
由此有Riemann和：
\begin{align*}
    \iint_D f(x,y)\dd{x}\dd{y} &= \sum_{i,j}f(\xi_{ij},\eta_{ij})\sigma(D_{ij}) \\
    &= \sum_{i,j}f(x(u_i,v_j),y(u_i,v_j))\left|\pdv{(x,y)}{(u,v)}\right|_{(u_i,v_j)}\Delta u_i \Delta v_j
\end{align*}
进而推出：
\begin{theorem}
    \textbf{二重积分的换元法}

    设$D$是$Oxy$平面中的可测有界闭区域，$f(x,y)\in C(D)$，当变换$(u,v)\mapsto(x,y)$是从$O'uv$平面上可测的有界区域$D'$到$Oxy$平面上$D$的有连续偏导数的1-1映射并满足$\displaystyle\pdv{(x,y)}{(u,v)}\neq 0$，则有：
    \[\iint_D f(x,y)\dd{x}\dd{y} = \iint_{D'}f(x(u,v),y(u,v))\left|\pdv{(x,y)}{(u,v)}\right|\dd{u}\dd{v}\]
    \textit{其中的Jacobi行列式可视为微元面积的缩放因子.特别对于直角坐标到极坐标的变换，这个因子为$r$.}
\end{theorem}

\begin{mistake}
    在以往的一元积分的换元法中，往往认为换元过程可以通过将换元表达式放入$\dd$中进行求导得到，例如将$x$换元为$x^2$有$2x\dd{x} = \dd{(x^2)}$.

    但是这个过程并不完全适用于重积分，因为一元积分所谓「求导」的过程实际上是乘上一个一阶的Jacobi行列式，恰好对应求一阶导.但对于二重或更高重的积分，求两个一阶导并不等价于相应阶数的Jacobi行列式，所以不能用一元积分的方法理解.

    有时候高阶的Jacobi行列式会退化为普通的一阶求导，比如在累次积分的过程中做单变量换元，那么仍可以用一元积分的方法.
\end{mistake}

关于二重积分换元法的严格证明，留待后续进行.

\begin{question}
    求椭球$\displaystyle\frac{x^2}{a^2} + \frac{y^2}{b^2} + \frac{z^2}{c^2} \leqslant 1$的体积$V$.
\end{question}

\begin{solution}
    椭球上半部分的函数为：
    \[z = c\sqrt{1-\frac{x^2}{a^2}-\frac{y^2}{b^2}},\quad (x,y)\in D = \left\{(x,y)\,\bigg|\,\frac{x^2}{a^2}+\frac{y^2}{b^2}\leqslant 1\right\}\]
    于是有：
    \[V = 2c \iint_D\sqrt{1-\frac{x^2}{a^2}-\frac{y^2}{b^2}}\dd{x}\dd{y}\]
    (椭圆$\to$圆)令$x \equiv as,y \equiv bt$，积分区域变为$D' = \{(s,t)\,|\,s^2+t^2\leqslant 1\}$，于是有：
    \[V = 2abc \iint_{D'}\sqrt{1-s^2-t^2}\dd{s}\dd{t}\]
    (圆$\to$矩形)令$s \equiv r\cos\theta,t \equiv r\sin\theta$，积分区域变为$D'' = \{(r,\theta)\,|\,r\in[0,1],\theta\in[0,2\pi]\}$，有：
    \[V = 2abc\iint_{D''}\sqrt{1-r^2}r\dd{r}\dd{\theta} = abc\int_{0}^{2\pi}\dd{\theta}\int_{0}^{1}\sqrt{1-r^2}\dd{r^2} = \frac{4}{3}\pi abc\]
\end{solution}

\begin{extension}{概率积分}
    反常积分$\displaystyle\int_{-\infty}^{+\infty}\ee^{-x^2}\dd{x}$没有原函数，难以求解.但是通过巧妙的构造，可以得出其结果.
    
    考虑求$\displaystyle\iint_{x^2+y^2\leqslant R^2}\ee^{-x^2-y^2}\dd{x}\dd{y}$：
    \[\iint_{x^2+y^2\leqslant R^2}\ee^{-x^2-y^2}\dd{x}\dd{y} = \int_{0}^{2\pi}\dd{\theta}\int_{0}^{R}\ee^{-r^2}r\dd{r} = \pi\left(1-\ee^{-R^2}\right)\]
    注意到：
    \[\left(\int_{-R}^{R}\ee^{-x^2}\dd{x}\right) = \int_{-R}^{R}\ee^{-x^2}\dd{x}\int_{-R}^{R}\ee^{-y^2}\dd{y} = \iint_{\,\substack{-R\leqslant x \leqslant R \\ -R\leqslant y \leqslant R}}\ee^{-x^2-y^2}\dd{x}\dd{y}\]
    显然上述积分区域是一个边长为$2R$的正方形，它有一个半径为$R$的内切圆和一个半径为$\sqrt{2}R$的外切圆.由于被积函数$\ee^{-x^2-y^2}$恒大于$0$，因而有如下不等式：
    \begin{gather*}
        \iint_{x^2+y^2\leqslant R^2}\ee^{-x^2-y^2}\dd{x}\dd{y} \leqslant \left(\int_{-R}^{R}\ee^{-x^2}\dd{x}\right)^2 \leqslant \iint_{x^2+y^2\leqslant 2R^2}\ee^{-x^2-y^2}\dd{x}\dd{y} \\
        \pi\left(1-\ee^{-R^2}\right) \leqslant \left(\int_{-R}^{R}\ee^{-x^2}\dd{x}\right)^2 \leqslant \pi\left(1-\ee^{-2R^2}\right)
    \end{gather*}
    若使$R \to \infty$，则根据三明治定理有：
    \[\lim_{R \to 0}\int_{-R}^{R}\ee^{-x^2}\dd{x} = \int_{-\infty}^{+\infty}\ee^{-x^2}\dd{x} = \sqrt{\pi}\]
    由此求得正态分布的概率积分：
    \[\int_{0}^{+\infty}\ee^{-x^2}\dd{x} = \frac{\sqrt{\pi}}{2}\]
\end{extension}

\begin{question}
    求$\displaystyle I = \iint_D\frac{\dd{x}\dd{y}}{xy(\ln^2 x + \ln^2 y)}$，其中$D$是在第一象限中由$x^2+y^2 = 1$和$x + y = 1$围成的区域.
\end{question}

\begin{solution}
    首先换元$x\equiv\ee^{r\cos\theta},y\equiv\ee^{r\sin\theta}$，计算其Jacobi行列式：
    \[\left|\pdv{(x,y)}{(r,\theta)}\right| = r\ee^{r\cos\theta + r\sin\theta}\]
    于是原式变为：
    \[I = \iint_{D'}\frac{1}{\ee^{r\cos\theta + r\sin\theta} r^2}r\ee^{r\cos\theta + r\sin\theta}\dd{r}\dd{\theta} = \iint_{D'}\frac{1}{r}\dd{r}\dd{\theta}\]
    接下来考察积分区域的变化.其中直线$x+y-1=0$变为一条隐式曲线$\ee^{r\cos\theta} + \ee^{r\sin\theta} - 1 = 0$.由于：
    \[x,y\in[0,1] \quad\Rightarrow\quad \cos\theta\leqslant0,\quad \sin\theta\leqslant 0 \quad\Rightarrow\quad \theta\in\left[\pi,\frac{3}{2}\pi\right]\]
    同时总是有$\pdv{r}\left(\ee^{r\cos\theta} + \ee^{r\sin\theta} - 1\right) \neq 0$，根据隐函数存在定理有$r = \varphi(\theta)$.

    积分区域的另一个边界$x^2+y^2-1=0$也被映射为一条隐式曲线$\ee^{2r\cos\theta} + \ee^{2r\sin\theta} - 1 = 0$，与直线同理，也推出$\theta$具有与直线相同的定义域，并且存在隐函数$r = \psi(\theta)$.

    即新的积分区域$D'$是$O'r\theta$平面上由两条平行于$r$轴的直线$\theta = \pi,\theta=\frac{3}{2}\pi$和两条隐式曲线$\varphi(\theta),\psi(\theta)$围成的区域.

    特别地，对于形如$\ee^{R\cos\theta}+\ee^{R\sin\theta}-1=0$的方程，其隐函数此前定义为$R = \varphi(\theta)$.对第一条曲线$R=r$，对第二条曲线，$R = 2r$，因而有$2r = \varphi(\theta) = 2\psi(\theta)$.注意到有$\psi(\theta) < \varphi(\theta)$，于是积分进一步写为：
    \[I = \int_{\pi}^{\frac{3}{2}\pi}\dd{\theta}\int_{\psi(\theta)}^{\varphi(\theta)}\frac{1}{r}\dd{r} = \ln r\Big|_{\psi(\theta)}^{2\psi(\theta)}\cdot\pi = \frac{\pi}{2}\ln 2\]
\end{solution}


\section{三重积分}
三重积分与二重积分无本质区别，构造Riemann和的方法也一致.性质如\textbf{可积必有界}、\textbf{连续必可积}等也与后者保持一致.


\subsection{三重积分的累次积分}
设$f(x,y,z)$在$V = I_1\times I_2 \times I_3 = [a_1,b_1]\times[a_2,b_2]\times[a_3,b_3]$上连续，采用如下分割：
\begin{gather*}
    T_x:a_1 = x_0 < x_1 < \cdots < x_n = b_1 \\
    T_y:a_2 = y_0 < y_1 < \cdots < y_m = b_2 \\
    T_z:a_3 = z_0 < z_1 < \cdots < z_k = b_3
\end{gather*}
由此构造Riemann和：
\[S(T) = \sum_{i=1}^{n}\sum_{j=1}^{m}\sum_{l=1}^{k}f(\xi_i,\eta_j,\zeta_l)\Delta x_i \Delta y_j \Delta z_l 
= \sum_{i=1}^{n}\sum_{j=1}^{m}\left[\sum_{l=1}^{k}f(\xi_i,\eta_j,\zeta_l)\Delta z_l\right]\Delta x_i \Delta y_j\]
同二次积分一样，在$\left\|T_z\right\| \to 0$时，括号内的式子可写为积分，并转为另外两个变量的函数：
\[\lim_{\left\|T_z\right\|\to0}\sum_{l=1}^{k}f(\xi_i,\eta_j,\zeta_l)\Delta z_l = \int_{a_3}^{b_3}f(\xi_i,\eta_j,z)\dd{z} = \varphi(\xi_i,\eta_j)\]
$S(T)$中剩余的两个求和号也基于同样的方法转换为积分，最终得到：
\[\iiint_V f(x,y,z)\dd{x}\dd{y}\dd{z} = \int_{a_1}^{b_1}\dd{x}\int_{a_2}^{b_2}\dd{y}\int_{a_3}^{b_3}f(x,y,z)\dd{z}\]
因为此前假设的连续性，积分次序可以交换，同时积分区域具有可加性.


\subsection{三重积分的换元}
依旧与二重积分一致的推导过程，变换$(u,v,w)\mapsto(x,y,z)$将$O'uvw$空间中的区域1-1映射到$Oxyz$空间中.坐标变换导致的微元体积变化依然可以用Jacobi行列式作为缩放因子：
\[\dd{V} = \dd{x}\dd{y}\dd{z} = \left|\pdv{(x,y,z)}{(u,v,w)}\right|\dd{u}\dd{v}\dd{w}\]
由此有：
\begin{theorem}
    \textbf{三重积分的换元法}
    \[\iiint_V\!\! f(x,y,z)\dd{x}\dd{y}\dd{z} =\! \iiint_{V'}\!\!\!f(x(u,v,w),y(u,v,w),z(u,v,w))\left|\pdv{(x,y,z)}{(u,v,w)}\right|\dd{u}\dd{v}\dd{w}\]
    \textit{特别地，对于直角坐标系到球坐标系的变换，其Jacobi行列式为$r^2 \sin\theta$.}
\end{theorem}

\begin{question}
    计算：$\displaystyle I = \iiint_{[0,1]^3}\frac{\dd{u}\dd{v}\dd{w}}{(1+u^2+v^2+w^2)^2}$.
\end{question}

\begin{solution}
    首先，观察积分区域是一个正方体，考虑累次积分；而后观察被积函数含有平方项，考虑球坐标.为了将二者结合起来，首先观察对称性.注意到，积分区域可以只取半个沿对角面切开的三棱柱：
    \[I = 2\iiint_V\frac{\dd{u}\dd{v}\dd{w}}{(1+u^2+v^2+w^2)^2},\quad V = \{(u,v,w)\,|\, 0\leqslant v \leqslant u \leqslant 1, 0\leqslant w \leqslant 1\}\]
    先转换到柱坐标，令$u\equiv r\cos\theta, v\equiv r\sin\theta, w\equiv w$，积分区域发生如下变化：
    \[0\leqslant r\sin\theta \leqslant r\cos\theta \leqslant 1 \quad\Rightarrow\quad 0 \leqslant \theta \leqslant \frac{\pi}{4},\quad 0\leqslant r \leqslant \frac{1}{\cos\theta} = \sec\theta\]
    由此得到：
    \[I = 2\iiint_{V'}\!\frac{r \dd{r}\dd{\theta}\dd{w}}{(1+w^2+r^2)^2},\quad V' = \left\{(r,\theta,w)\,\Big|\, 0\leqslant r \leqslant \sec\theta, 0\leqslant\theta\leqslant\frac{\pi}{4},0\leqslant w \leqslant 1\right\}\]
    而后转换至球坐标，令$w \equiv \tan\varphi$，由此有$0\leqslant\varphi\leqslant\dfrac{\pi}{4},\dd{w} = \sec^2\varphi\dd{\varphi}$：
    \begin{gather*}
        I = 2\iiint_{V''}\frac{r\sec^2\varphi\dd{r}\dd{\theta}\dd{\varphi}}{(1+r^2+\tan^2\varphi)^2} = 2\iiint_{V''}\frac{r\sec^2\varphi\dd{r}\dd{\theta}\dd{\varphi}}{(r^2+\sec^2\varphi)^2} \\
        V'' = \left\{(r,\theta,\varphi)\,\Big|\,0\leqslant r \leqslant\sec\theta,0\leqslant\theta\leqslant\frac{\pi}{4},0\leqslant\varphi\leqslant\frac{\pi}{4}\right\}
    \end{gather*}
    由于$r$的上限中包含有变量$\theta$，因此先对$r$积分：
    \begin{align*}
        I &= 2\iint_{[0,\frac{\pi}{4}]^2}\dd{\theta}\dd{\varphi}\int_{0}^{\sec\theta}\frac{r\sec^2\varphi}{(r^2+\sec^2\varphi)^2}\dd{r} \\
        &= \iint_{[0,\frac{\pi}{4}]^2}\sec^2\varphi\dd{\theta}\dd{\varphi}\int_{0}^{\sec\theta}\frac{\dd{(r^2)}}{(r^2+\sec^2\varphi)^2} \\
        &= - \iint_{[0,\frac{\pi}{4}]^2}\sec^2\varphi\dd{\theta}\dd{\varphi} \cdot \left.\frac{1}{r^2+\sec^2\varphi}\right|_{r=0}^{r=\sec\theta} \\
        &= \iint_{[0,\frac{\pi}{4}]^2}\left(1-\frac{\sec^2\varphi}{\sec^2\theta+\sec^2\varphi}\right)\dd{\theta}\dd{\varphi} \\
        &= \left(\frac{\pi}{4}\right)^2 - \iint_{[0,\frac{\pi}{4}]^2}\frac{\sec^2\varphi}{\sec^2\theta+\sec^2\varphi}\dd{\theta}\dd{\varphi}
    \end{align*}
    上式中，记$\displaystyle A = \iint_{[0,\frac{\pi}{4}]^2}\frac{\sec^2\varphi}{\sec^2\theta+\sec^2\varphi}\dd{\theta}\dd{\varphi}$，根据$\varphi$和$\theta$的对称性则有：
    \begin{gather*}
        A = \iint_{[0,\frac{\pi}{4}]^2}\frac{\sec^2\varphi}{\sec^2\theta+\sec^2\varphi}\dd{\theta}\dd{\varphi} = \iint_{[0,\frac{\pi}{4}]^2}\frac{\sec^2\theta}{\sec^2\varphi+\sec^2\theta}\dd{\theta}\dd{\varphi} \\
        2A = \iint_{[0,\frac{\pi}{4}]^2}\frac{\sec^2\varphi}{\sec^2\theta+\sec^2\varphi}\dd{\theta}\dd{\varphi} + \iint_{[0,\frac{\pi}{4}]^2}\frac{\sec^2\theta}{\sec^2\varphi+\sec^2\theta}\dd{\theta}\dd{\varphi} = \iint_{[0,\frac{\pi}{4}]^2}1\dd{\theta}\dd{\varphi} \\
        \Rightarrow\quad A = \frac{1}{2}\left(\frac{\pi}{4}\right)^2
    \end{gather*}
    最终得到$\displaystyle I = \frac{\pi^2}{32}$.
\end{solution}
